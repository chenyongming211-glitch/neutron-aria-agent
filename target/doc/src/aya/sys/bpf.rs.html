39pub(crate) fn bpf_create_map(
40    name: &CStr,
41    def: &obj::Map,
42    btf_fd: Option<BorrowedFd<'_>>,
43    kernel_version: KernelVersion,
44) -> SysResult<crate::MockableFd> {
45    let mut attr = unsafe { mem::zeroed::<bpf_attr>() };
46
47    let u = unsafe { &mut attr.__bindgen_anon_1 };
48    u.map_type = def.map_type();
49    u.key_size = def.key_size();
50    u.value_size = def.value_size();
51    u.max_entries = def.max_entries();
52    u.map_flags = def.map_flags();
53
54    if let obj::Map::Btf(m) = def {
55        use bpf_map_type::*;
56
57        // Mimic https://github.com/libbpf/libbpf/issues/355
58        // Currently a bunch of (usually pretty specialized) BPF maps do not support
59        // specifying BTF types for the key and value.
60        match u.map_type.try_into() {
61            Ok(BPF_MAP_TYPE_PERF_EVENT_ARRAY)
62            | Ok(BPF_MAP_TYPE_CGROUP_ARRAY)
63            | Ok(BPF_MAP_TYPE_STACK_TRACE)
64            | Ok(BPF_MAP_TYPE_ARRAY_OF_MAPS)
65            | Ok(BPF_MAP_TYPE_HASH_OF_MAPS)
66            | Ok(BPF_MAP_TYPE_DEVMAP)
67            | Ok(BPF_MAP_TYPE_DEVMAP_HASH)
68            | Ok(BPF_MAP_TYPE_CPUMAP)
69            | Ok(BPF_MAP_TYPE_XSKMAP)
70            | Ok(BPF_MAP_TYPE_SOCKMAP)
71            | Ok(BPF_MAP_TYPE_SOCKHASH)
72            | Ok(BPF_MAP_TYPE_QUEUE)
73            | Ok(BPF_MAP_TYPE_STACK)
74            | Ok(BPF_MAP_TYPE_RINGBUF) => {
75                u.btf_key_type_id = 0;
76                u.btf_value_type_id = 0;
77                u.btf_fd = 0;
78            }
79            _ => {
80                u.btf_key_type_id = m.def.btf_key_type_id;
81                u.btf_value_type_id = m.def.btf_value_type_id;
82                u.btf_fd = btf_fd.map(|fd| fd.as_raw_fd()).unwrap_or_default() as u32;
83            }
84        }
85    }
86
87    // https://github.com/torvalds/linux/commit/ad5b177bd73f5107d97c36f56395c4281fb6f089
88    // The map name was added as a parameter in kernel 4.15+ so we skip adding it on
89    // older kernels for compatibility
90    if kernel_version >= KernelVersion::new(4, 15, 0) {
91        // u.map_name is 16 bytes max and must be NULL terminated
92        let name_len = cmp::min(name.to_bytes().len(), BPF_OBJ_NAME_LEN - 1);
93        u.map_name[..name_len]
94            .copy_from_slice(unsafe { slice::from_raw_parts(name.as_ptr(), name_len) });
95    }
96
97    // SAFETY: BPF_MAP_CREATE returns a new file descriptor.
98    unsafe { fd_sys_bpf(bpf_cmd::BPF_MAP_CREATE, &mut attr) }
99}

136pub(crate) fn bpf_load_program(
137    aya_attr: &EbpfLoadProgramAttrs<'_>,
138    log_buf: &mut [u8],
139    verifier_log_level: VerifierLogLevel,
140) -> SysResult<crate::MockableFd> {
141    let mut attr = unsafe { mem::zeroed::<bpf_attr>() };
142
143    let u = unsafe { &mut attr.__bindgen_anon_3 };
144
145    if let Some(prog_name) = &aya_attr.name {
146        let mut name: [c_char; 16] = [0; 16];
147        let name_bytes = prog_name.to_bytes();
148        let len = cmp::min(name.len(), name_bytes.len());
149        name[..len].copy_from_slice(unsafe {
150            slice::from_raw_parts(name_bytes.as_ptr() as *const c_char, len)
151        });
152        u.prog_name = name;
153    }
154
155    u.prog_flags = aya_attr.flags;
156    u.prog_type = aya_attr.ty as u32;
157    if let Some(v) = aya_attr.expected_attach_type {
158        u.expected_attach_type = v as u32;
159    }
160    u.insns = aya_attr.insns.as_ptr() as u64;
161    u.insn_cnt = aya_attr.insns.len() as u32;
162    u.license = aya_attr.license.as_ptr() as u64;
163    u.kern_version = aya_attr.kernel_version;
164
165    // these must be allocated here to ensure the slice outlives the pointer
166    // so .as_ptr below won't point to garbage
167    let line_info_buf = aya_attr.line_info.line_info_bytes();
168    let func_info_buf = aya_attr.func_info.func_info_bytes();
169
170    if let Some(btf_fd) = aya_attr.prog_btf_fd {
171        u.prog_btf_fd = btf_fd.as_raw_fd() as u32;
172        if aya_attr.line_info_rec_size > 0 {
173            u.line_info = line_info_buf.as_ptr() as *const _ as u64;
174            u.line_info_cnt = aya_attr.line_info.len() as u32;
175            u.line_info_rec_size = aya_attr.line_info_rec_size as u32;
176        }
177        if aya_attr.func_info_rec_size > 0 {
178            u.func_info = func_info_buf.as_ptr() as *const _ as u64;
179            u.func_info_cnt = aya_attr.func_info.len() as u32;
180            u.func_info_rec_size = aya_attr.func_info_rec_size as u32;
181        }
182    }
183    if !log_buf.is_empty() {
184        u.log_level = verifier_log_level.bits();
185        u.log_buf = log_buf.as_mut_ptr() as u64;
186        u.log_size = log_buf.len() as u32;
187    }
188    if let Some(v) = aya_attr.attach_btf_obj_fd {
189        u.__bindgen_anon_1.attach_btf_obj_fd = v.as_raw_fd() as _;
190    }
191    if let Some(v) = aya_attr.attach_prog_fd {
192        u.__bindgen_anon_1.attach_prog_fd = v.as_raw_fd() as u32;
193    }
194
195    if let Some(v) = aya_attr.attach_btf_id {
196        u.attach_btf_id = v;
197    }
198    bpf_prog_load(&mut attr)
199}

201fn lookup<K: Pod, V: Pod>(
202    fd: BorrowedFd<'_>,
203    key: Option<&K>,
204    flags: u64,
205    cmd: bpf_cmd,
206) -> SysResult<Option<V>> {
207    let mut attr = unsafe { mem::zeroed::<bpf_attr>() };
208    let mut value = MaybeUninit::zeroed();
209
210    let u = unsafe { &mut attr.__bindgen_anon_2 };
211    u.map_fd = fd.as_raw_fd() as u32;
212    if let Some(key) = key {
213        u.key = key as *const _ as u64;
214    }
215    u.__bindgen_anon_1.value = &mut value as *mut _ as u64;
216    u.flags = flags;
217
218    match sys_bpf(cmd, &mut attr) {
219        Ok(_) => Ok(Some(unsafe { value.assume_init() })),
220        Err((_, io_error)) if io_error.raw_os_error() == Some(ENOENT) => Ok(None),
221        Err(e) => Err(e),
222    }
223}

1155fn bpf_obj_get_next_id(
1156    id: u32,
1157    cmd: bpf_cmd,
1158    name: &'static str,
1159) -> Result<Option<u32>, SyscallError> {
1160    let mut attr = unsafe { mem::zeroed::<bpf_attr>() };
1161    let u = unsafe { &mut attr.__bindgen_anon_6 };
1162    u.__bindgen_anon_1.start_id = id;
1163    match sys_bpf(cmd, &mut attr) {
1164        Ok(code) => {
1165            assert_eq!(code, 0);
1166            Ok(Some(unsafe { attr.__bindgen_anon_6.next_id }))
1167        }
1168        Err((code, io_error)) => {
1169            assert_eq!(code, -1);

1171                Ok(None)
1172            } else {
1173                Err(SyscallError {
1174                    call: name,
1175                    io_error,
1176                })

1178        }
1179    }

1182fn iter_obj_ids(
1183    cmd: bpf_cmd,
1184    name: &'static str,
1185) -> impl Iterator<Item = Result<u32, SyscallError>> {
1186    let mut current_id = Some(0);
1187    iter::from_fn(move || {
1188        let next_id = {
1189            let current_id = current_id?;
1190            bpf_obj_get_next_id(current_id, cmd, name).transpose()
1191        };
1192        current_id = next_id.as_ref().and_then(|next_id| match next_id {
1193            Ok(next_id) => Some(*next_id),
1194            Err(SyscallError { .. }) => None,
1195        });
1196        next_id
1197    })
1198}

1230pub(crate) fn retry_with_verifier_logs<T>(
1231    max_retries: usize,
1232    f: impl Fn(&mut [u8]) -> SysResult<T>,
1233) -> (SysResult<T>, VerifierLog) {
1234    const MIN_LOG_BUF_SIZE: usize = 1024 * 10;
1235    const MAX_LOG_BUF_SIZE: usize = (u32::MAX >> 8) as usize;
1236
1237    let mut log_buf = Vec::new();
1238    let mut retries = 0;
1239    loop {
1240        let ret = f(log_buf.as_mut_slice());
1241        if retries != max_retries {
1242            if let Err((_, io_error)) = &ret {
1243                if retries == 0 || io_error.raw_os_error() == Some(ENOSPC) {
1244                    let len = (log_buf.capacity() * 10).clamp(MIN_LOG_BUF_SIZE, MAX_LOG_BUF_SIZE);
1245                    log_buf.resize(len, 0);
1246                    if let Some(first) = log_buf.first_mut() {
1247                        *first = 0;
1248                    }
1249                    retries += 1;
1250                    continue;
1251                }
1252            }
1253        }
1254        if let Some(pos) = log_buf.iter().position(|b| *b == 0) {
1255            log_buf.truncate(pos);
1256        }
1257        let log_buf = String::from_utf8(log_buf).unwrap();
1258
1259        break (ret, VerifierLog::new(log_buf));
1260    }
1261}
