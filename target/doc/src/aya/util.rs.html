22/// Represents a kernel version, in major.minor.release version.
23// Adapted from https://docs.rs/procfs/latest/procfs/sys/kernel/struct.Version.html.
24#[derive(Debug, Copy, Clone, Eq, PartialEq, PartialOrd)]
25pub struct KernelVersion {
26    pub(crate) major: u8,
27    pub(crate) minor: u8,
28    pub(crate) patch: u16,
29}
30
31#[derive(thiserror::Error, Debug)]
32enum CurrentKernelVersionError {
33    #[error("failed to read kernel version")]
34    IO(#[from] io::Error),
35    #[error("failed to parse kernel version")]
36    ParseError(String),
37    #[error("kernel version string is not valid UTF-8")]
38    Utf8(#[from] Utf8Error),
39}
40
41impl KernelVersion {
42    /// Constructor.
43    pub fn new(major: u8, minor: u8, patch: u16) -> Self {
44        Self {
45            major,
46            minor,
47            patch,
48        }
49    }
50
51    /// Returns the kernel version of the currently running kernel.
52    pub fn current() -> Result<Self, impl Error> {
53        Self::get_kernel_version()
54    }
55
56    /// The equivalent of LINUX_VERSION_CODE.
57    pub fn code(self) -> u32 {
58        let Self {
59            major,
60            minor,
61            mut patch,
62        } = self;
63
64        // Certain LTS kernels went above the "max" 255 patch so
65        // backports were done to cap the patch version
66        let max_patch = match (major, minor) {
67            // On 4.4 + 4.9, any patch 257 or above was hardcoded to 255.
68            // See: https://github.com/torvalds/linux/commit/a15813a +
69            // https://github.com/torvalds/linux/commit/42efb098
70            (4, 4 | 9) => 257,
71            // On 4.14, any patch 252 or above was hardcoded to 255.
72            // See: https://github.com/torvalds/linux/commit/e131e0e
73            (4, 14) => 252,
74            // On 4.19, any patch 222 or above was hardcoded to 255.
75            // See: https://github.com/torvalds/linux/commit/a256aac
76            (4, 19) => 222,
77            // For other kernels (i.e., newer LTS kernels as other
78            // ones won't reach 255+ patches) clamp it to 255. See:
79            // https://github.com/torvalds/linux/commit/9b82f13e
80            _ => 255,
81        };
82
83        // anything greater or equal to `max_patch` is hardcoded to
84        // 255.
85        if patch >= max_patch {
86            patch = 255;
87        }
88
89        (u32::from(major) << 16) + (u32::from(minor) << 8) + u32::from(patch)
90    }
91
92    // These (get_ubuntu_kernel_version, parse_ubuntu_kernel_version, read_ubuntu_kernel_version_file)
93    // are ported from https://github.com/torvalds/linux/blob/3f01e9f/tools/lib/bpf/libbpf_probes.c#L21-L101.
94    fn get_ubuntu_kernel_version() -> Result<Option<Self>, CurrentKernelVersionError> {
95        Self::read_ubuntu_kernel_version_file().and_then(|content| {
96            content
97                .and_then(|content| Self::parse_ubuntu_kernel_version(&content).transpose())
98                .transpose()
99        })
100    }
101
102    fn read_ubuntu_kernel_version_file() -> Result<Option<String>, CurrentKernelVersionError> {
103        const UBUNTU_KVER_FILE: &str = "/proc/version_signature";
104        match fs::read_to_string(UBUNTU_KVER_FILE) {
105            Ok(s) => Ok(Some(s)),
106            Err(e) => {
107                if e.kind() != io::ErrorKind::NotFound {
108                    Err(e.into())
109                } else {
110                    Ok(None)
111                }
112            }
113        }
114    }
115
116    fn parse_ubuntu_kernel_version(s: &str) -> Result<Option<Self>, CurrentKernelVersionError> {
117        let mut parts = s.split_terminator(char::is_whitespace);
118        let mut next = || {
119            parts
120                .next()
121                .ok_or_else(|| CurrentKernelVersionError::ParseError(s.to_string()))
122        };
123        let _ubuntu: &str = next()?;
124        let _ubuntu_version: &str = next()?;
125        let kernel_version_string = next()?;
126        Self::parse_kernel_version_string(kernel_version_string).map(Some)
127    }
128
129    fn get_debian_kernel_version(
130        info: &utsname,
131    ) -> Result<Option<Self>, CurrentKernelVersionError> {
132        // Safety: man 2 uname:
133        //
134        // The length of the arrays in a struct utsname is unspecified (see NOTES); the fields are
135        // terminated by a null byte ('\0').
136        let s = unsafe { CStr::from_ptr(info.version.as_ptr()) };
137        let s = s.to_str()?;
138        let kernel_version_string = match s.split_once("Debian ") {
139            Some((_prefix, suffix)) => suffix,
140            None => return Ok(None),
141        };
142        Self::parse_kernel_version_string(kernel_version_string).map(Some)
143    }
144
145    fn get_kernel_version() -> Result<Self, CurrentKernelVersionError> {
146        if let Ok(Some(v)) = Self::get_ubuntu_kernel_version() {
147            return Ok(v);
148        }
149
150        let mut info = unsafe { mem::zeroed::<utsname>() };
151        if unsafe { uname(&mut info) } != 0 {
152            return Err(io::Error::last_os_error().into());
153        }
154
155        if let Some(v) = Self::get_debian_kernel_version(&info)? {
156            return Ok(v);
157        }
158
159        // Safety: man 2 uname:
160        //
161        // The length of the arrays in a struct utsname is unspecified (see NOTES); the fields are
162        // terminated by a null byte ('\0').
163        let s = unsafe { CStr::from_ptr(info.release.as_ptr()) };
164        let s = s.to_str()?;
165        Self::parse_kernel_version_string(s)
166    }
167
168    fn parse_kernel_version_string(s: &str) -> Result<Self, CurrentKernelVersionError> {
169        fn parse<T: FromStr<Err = ParseIntError>>(s: Option<&str>) -> Option<T> {
170            s.map(str::parse).transpose().unwrap_or_default()
171        }
172        let error = || CurrentKernelVersionError::ParseError(s.to_string());
173        let mut parts = s.split(|c: char| c == '.' || !c.is_ascii_digit());
174        let major = parse(parts.next()).ok_or_else(error)?;
175        let minor = parse(parts.next()).ok_or_else(error)?;
176        let patch = parse(parts.next()).ok_or_else(error)?;
177        Ok(Self::new(major, minor, patch))
178    }
179}

183        write!(f, "{}.{}.{}", self.major, self.minor, self.patch)
184    }
185}

187const ONLINE_CPUS: &str = "/sys/devices/system/cpu/online";
188const POSSIBLE_CPUS: &str = "/sys/devices/system/cpu/possible";

190/// Returns the numeric IDs of the CPUs currently online.
191pub fn online_cpus() -> Result<Vec<u32>, (&'static str, io::Error)> {
192    read_cpu_ranges(ONLINE_CPUS)
193}

195/// Get the number of possible cpus.
196///
197/// See `/sys/devices/system/cpu/possible`.
198pub fn nr_cpus() -> Result<usize, (&'static str, io::Error)> {
199    thread_local! {
200        // TODO(https://github.com/rust-lang/rust/issues/109737): Use
201        // `std::cell::OnceCell` when `get_or_try_init` is stabilized.
202        static CACHE: once_cell::unsync::OnceCell<usize> = const { once_cell::unsync::OnceCell::new() };
203    }
204    CACHE.with(|cell| {
205        cell.get_or_try_init(|| read_cpu_ranges(POSSIBLE_CPUS).map(|cpus| cpus.len()))
206            .copied()
207    })
208}
209
210fn read_cpu_ranges(path: &'static str) -> Result<Vec<u32>, (&'static str, io::Error)> {
211    (|| {
212        let data = fs::read_to_string(path)?;
213        parse_cpu_ranges(data.trim())
214    })()
215    .map_err(|error| (path, error))
216}
217
218fn parse_cpu_ranges(data: &str) -> Result<Vec<u32>, io::Error> {
219    data.split(',')
220        .map(|range| {
221            let mut iter = range
222                .split('-')
223                .map(|s| s.parse::<u32>().map_err(|ParseIntError { .. }| range));
224            let start = iter.next().unwrap()?; // str::split always returns at least one element.
225            let end = match iter.next() {
226                None => start,
227                Some(end) => {
228                    if iter.next().is_some() {
229                        return Err(range);
230                    }
231                    end?
232                }
233            };
234            Ok(start..=end)
235        })
236        .try_fold(Vec::new(), |mut cpus, range| {
237            let range = range.map_err(|range| io::Error::new(io::ErrorKind::InvalidData, range))?;
238            cpus.extend(range);
239            Ok(cpus)
240        })
241}

243/// Loads kernel symbols from `/proc/kallsyms`.
244///
245/// See [`crate::maps::StackTraceMap`] for an example on how to use this to resolve kernel addresses to symbols.
246pub fn kernel_symbols() -> Result<BTreeMap<u64, String>, io::Error> {
247    let mut reader = BufReader::new(File::open("/proc/kallsyms")?);
248    parse_kernel_symbols(&mut reader)
249}
250
251fn parse_kernel_symbols(reader: impl BufRead) -> Result<BTreeMap<u64, String>, io::Error> {
252    reader
253        .lines()
254        .map(|line| {
255            let line = line?;
256            (|| {
257                let mut parts = line.splitn(4, ' ');
258                let addr = parts.next()?;
259                let _kind = parts.next()?;
260                let name = parts.next()?;
261                let addr = match u64::from_str_radix(addr, 16) {
262                    Ok(addr) => Some(addr),
263                    Err(ParseIntError { .. }) => None,
264                }?;
265                Some((addr, name.to_owned()))
266            })()
267            .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, line.clone()))
268        })
269        .collect()
270}

272/// Returns the prefix used by syscalls.
273///
274/// # Example
275///
276/// ```no_run
277/// use aya::util::syscall_prefix;
278/// let prefix = syscall_prefix().unwrap();
279/// let syscall_fname = format!("{prefix}exec");
280/// ```
281///
282/// # Errors
283///
284/// Returns [`std::io::ErrorKind::NotFound`] if the prefix can't be guessed. Returns other [`std::io::Error`] kinds if `/proc/kallsyms` can't be opened or is somehow invalid.
285#[deprecated(
286    since = "0.12.0",
287    note = "On some systems - commonly on 64 bit kernels that support running \
288    32 bit applications - the syscall prefix depends on what architecture an \
289    application is compiled for, therefore attaching to only one prefix is \
290    incorrect and can lead to security issues."
291)]
292pub fn syscall_prefix() -> Result<&'static str, io::Error> {
293    const PREFIXES: [&str; 7] = [
294        "sys_",
295        "__x64_sys_",
296        "__x32_compat_sys_",
297        "__ia32_compat_sys_",
298        "__arm64_sys_",
299        "__s390x_sys_",
300        "__s390_sys_",
301    ];
302    let ksym = kernel_symbols()?;
303    for p in PREFIXES {
304        let prefixed_syscall = format!("{}bpf", p);
305        if ksym.values().any(|el| *el == prefixed_syscall) {
306            return Ok(p);
307        }
308    }
309    Err(io::ErrorKind::NotFound.into())
310}

312pub(crate) fn ifindex_from_ifname(if_name: &str) -> Result<u32, io::Error> {
313    let c_str_if_name = CString::new(if_name)?;
314    let c_if_name = c_str_if_name.as_ptr();
315    // Safety: libc wrapper
316    let if_index = unsafe { if_nametoindex(c_if_name) };
317    if if_index == 0 {
318        return Err(io::Error::last_os_error());
319    }
320    Ok(if_index)
321}

327/// Include bytes from a file for use in a subsequent [`crate::Ebpf::load`].
328///
329/// This macro differs from the standard `include_bytes!` macro since it also ensures that
330/// the bytes are correctly aligned to be parsed as an ELF binary. This avoid some nasty
331/// compilation errors when the resulting byte array is not the correct alignment.
332///
333/// # Examples
334/// ```ignore
335/// use aya::{Ebpf, include_bytes_aligned};
336///
337/// let mut bpf = Ebpf::load(include_bytes_aligned!(
338///     "/path/to/bpf.o"
339/// ))?;
340///
341/// # Ok::<(), aya::EbpfError>(())
342/// ```
343#[macro_export]
344macro_rules! include_bytes_aligned {
345    ($path:expr) => {{
346        #[repr(align(32))]
347        pub struct Aligned32;
348
349        #[repr(C)]
350        pub struct Aligned<Bytes: ?Sized> {
351            pub _align: [Aligned32; 0],
352            pub bytes: Bytes,
353        }
354
355        const ALIGNED: &Aligned<[u8]> = &Aligned {
356            _align: [],
357            bytes: *include_bytes!($path),
358        };
359
360        &ALIGNED.bytes
361    }};
362}

369// bytes_of converts a <T> to a byte slice
370pub(crate) unsafe fn bytes_of<T: Pod>(val: &T) -> &[u8] {
371    let size = mem::size_of::<T>();
372    slice::from_raw_parts(slice::from_ref(val).as_ptr().cast(), size)
373}
374
375pub(crate) fn bytes_of_slice<T: Pod>(val: &[T]) -> &[u8] {
376    let size = val.len().wrapping_mul(mem::size_of::<T>());
377    // Safety:
378    // Any alignment is allowed.
379    // The size is determined in this function.
380    // The Pod trait ensures the type is valid to cast to bytes.
381    unsafe { slice::from_raw_parts(val.as_ptr().cast(), size) }
382}

384pub(crate) fn bytes_of_bpf_name(bpf_name: &[core::ffi::c_char; 16]) -> &[u8] {
385    let length = bpf_name
386        .iter()
387        .rposition(|ch| *ch != 0)
388        .map(|pos| pos + 1)
389        .unwrap_or(0);
390    unsafe { slice::from_raw_parts(bpf_name.as_ptr() as *const _, length) }
391}
