12/// A program that attaches to Linux LSM hooks. Used to implement security policy and
13/// audit logging.
14///
15/// LSM probes can be attached to the kernel's [security hooks][1] to implement mandatory
16/// access control policy and security auditing.
17///
18/// LSM probes require a kernel compiled with `CONFIG_BPF_LSM=y` and `CONFIG_DEBUG_INFO_BTF=y`.
19/// In order for the probes to fire, you also need the BPF LSM to be enabled through your
20/// kernel's boot paramters (like `lsm=lockdown,yama,bpf`).
21///
22/// # Minimum kernel version
23///
24/// The minimum kernel version required to use this feature is 5.7.
25///
26/// # Examples
27///
28/// ```no_run
29/// # #[derive(thiserror::Error, Debug)]
30/// # enum LsmError {
31/// #     #[error(transparent)]
32/// #     BtfError(#[from] aya::BtfError),
33/// #     #[error(transparent)]
34/// #     Program(#[from] aya::programs::ProgramError),
35/// #     #[error(transparent)]
36/// #     Ebpf(#[from] aya::EbpfError),
37/// # }
38/// # let mut bpf = Ebpf::load_file("ebpf_programs.o")?;
39/// use aya::{Ebpf, programs::Lsm, BtfError, Btf};
40///
41/// let btf = Btf::from_sys_fs()?;
42/// let program: &mut Lsm = bpf.program_mut("lsm_prog").unwrap().try_into()?;
43/// program.load("security_bprm_exec", &btf)?;
44/// program.attach()?;
45/// # Ok::<(), LsmError>(())
46/// ```
47///
48/// [1]: https://elixir.bootlin.com/linux/latest/source/include/linux/lsm_hook_defs.h
49#[derive(Debug)]
50#[doc(alias = "BPF_PROG_TYPE_LSM")]
51pub struct Lsm {
52    pub(crate) data: ProgramData<LsmLink>,
53}
54
55impl Lsm {
56    /// Loads the program inside the kernel.
57    ///
58    /// # Arguments
59    ///
60    /// * `lsm_hook_name` - full name of the LSM hook that the program should
61    ///   be attached to
62    pub fn load(&mut self, lsm_hook_name: &str, btf: &Btf) -> Result<(), ProgramError> {
63        self.data.expected_attach_type = Some(BPF_LSM_MAC);
64        let type_name = format!("bpf_lsm_{lsm_hook_name}");
65        self.data.attach_btf_id =
66            Some(btf.id_by_type_name_kind(type_name.as_str(), BtfKind::Func)?);
67        load_program(BPF_PROG_TYPE_LSM, &mut self.data)
68    }
69
70    /// Attaches the program.
71    ///
72    /// The returned value can be used to detach, see [Lsm::detach].
73    pub fn attach(&mut self) -> Result<LsmLinkId, ProgramError> {
74        attach_raw_tracepoint(&mut self.data, None)
75    }
76
77    /// Detaches the program.
78    ///
79    /// See [Lsm::attach].
80    pub fn detach(&mut self, link_id: LsmLinkId) -> Result<(), ProgramError> {
81        self.data.links.remove(link_id)
82    }
83
84    /// Takes ownership of the link referenced by the provided link_id.
85    ///
86    /// The link will be detached on `Drop` and the caller is now responsible
87    /// for managing its lifetime.
88    pub fn take_link(&mut self, link_id: LsmLinkId) -> Result<LsmLink, ProgramError> {
89        self.data.take_link(link_id)
90    }
91}
