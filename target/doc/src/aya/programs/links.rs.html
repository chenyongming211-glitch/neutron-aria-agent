34/// Program attachment mode.
35#[derive(Clone, Copy, Debug, Default)]
36pub enum CgroupAttachMode {
37    /// Allows only one BPF program in the cgroup subtree.
38    #[default]
39    Single,
40
41    /// Allows the program to be overridden by one in a sub-cgroup.
42    AllowOverride,
43
44    /// Allows multiple programs to be run in the cgroup subtree.
45    AllowMultiple,
46}
47
48impl From<CgroupAttachMode> for u32 {
49    fn from(mode: CgroupAttachMode) -> Self {
50        match mode {
51            CgroupAttachMode::Single => 0,
52            CgroupAttachMode::AllowOverride => BPF_F_ALLOW_OVERRIDE,
53            CgroupAttachMode::AllowMultiple => BPF_F_ALLOW_MULTI,
54        }
55    }
56}

58#[derive(Debug)]
59pub(crate) struct LinkMap<T: Link> {
60    links: HashMap<T::Id, T>,
61}
62
63impl<T: Link> LinkMap<T> {
64    pub(crate) fn new() -> Self {
65        Self {
66            links: HashMap::new(),
67        }
68    }
69
70    pub(crate) fn insert(&mut self, link: T) -> Result<T::Id, ProgramError> {
71        let id = link.id();
72
73        match self.links.entry(link.id()) {
74            Entry::Occupied(_) => return Err(ProgramError::AlreadyAttached),
75            Entry::Vacant(e) => e.insert(link),
76        };
77
78        Ok(id)
79    }
80
81    pub(crate) fn remove(&mut self, link_id: T::Id) -> Result<(), ProgramError> {
82        self.links
83            .remove(&link_id)
84            .ok_or(ProgramError::NotAttached)?
85            .detach()
86    }
87
88    pub(crate) fn remove_all(&mut self) -> Result<(), ProgramError> {
89        for (_, link) in self.links.drain() {
90            link.detach()?;
91        }
92        Ok(())
93    }
94
95    pub(crate) fn forget(&mut self, link_id: T::Id) -> Result<T, ProgramError> {
96        self.links.remove(&link_id).ok_or(ProgramError::NotAttached)
97    }
98}
99
100impl<T: Link> Drop for LinkMap<T> {
101    fn drop(&mut self) {
102        let _ = self.remove_all();
103    }
104}

110/// A file descriptor link.
111///
112/// Fd links are returned directly when attaching some program types (for
113/// instance [`crate::programs::cgroup_skb::CgroupSkb`]), or can be obtained by
114/// converting other link types (see the `TryFrom` implementations).
115///
116/// An important property of fd links is that they can be pinned. Pinning
117/// can be used keep a link attached "in background" even after the program
118/// that has created the link terminates.
119///
120/// # Example
121///
122/// ```no_run
123/// # let mut bpf = Ebpf::load_file("ebpf_programs.o")?;
124/// use aya::{Ebpf, programs::{links::FdLink, KProbe}};
125///
126/// let program: &mut KProbe = bpf.program_mut("intercept_wakeups").unwrap().try_into()?;
127/// program.load()?;
128/// let link_id = program.attach("try_to_wake_up", 0)?;
129/// let link = program.take_link(link_id).unwrap();
130/// let fd_link: FdLink = link.try_into().unwrap();
131/// fd_link.pin("/sys/fs/bpf/intercept_wakeups_link").unwrap();
132///
133/// # Ok::<(), aya::EbpfError>(())
134/// ```
135#[derive(Debug)]
136pub struct FdLink {
137    pub(crate) fd: crate::MockableFd,
138}
139
140impl FdLink {
141    pub(crate) fn new(fd: crate::MockableFd) -> Self {
142        Self { fd }
143    }

145    /// Pins the link to a BPF file system.
146    ///
147    /// When a link is pinned it will remain attached even after the link instance is dropped,
148    /// and will only be detached once the pinned file is removed. To unpin, see [`PinnedLink::unpin()`].
149    ///
150    /// The parent directories in the provided path must already exist before calling this method,
151    /// and must be on a BPF file system (bpffs).
152    ///
153    /// # Example
154    /// ```no_run
155    /// # use aya::programs::{links::FdLink, Extension};
156    /// # use std::convert::TryInto;
157    /// # #[derive(thiserror::Error, Debug)]
158    /// # enum Error {
159    /// #     #[error(transparent)]
160    /// #     Ebpf(#[from] aya::EbpfError),
161    /// #     #[error(transparent)]
162    /// #     Pin(#[from] aya::pin::PinError),
163    /// #     #[error(transparent)]
164    /// #     Program(#[from] aya::programs::ProgramError)
165    /// # }
166    /// # let mut bpf = aya::Ebpf::load(&[])?;
167    /// # let prog: &mut Extension = bpf.program_mut("example").unwrap().try_into()?;
168    /// let link_id = prog.attach()?;
169    /// let owned_link = prog.take_link(link_id)?;
170    /// let fd_link: FdLink = owned_link.into();
171    /// let pinned_link = fd_link.pin("/sys/fs/bpf/example")?;
172    /// # Ok::<(), Error>(())
173    /// ```
174    pub fn pin<P: AsRef<Path>>(self, path: P) -> Result<PinnedLink, PinError> {
175        use std::os::unix::ffi::OsStrExt as _;
176
177        let path = path.as_ref();
178        let path_string = CString::new(path.as_os_str().as_bytes()).map_err(|error| {
179            PinError::InvalidPinPath {
180                path: path.into(),
181                error,
182            }
183        })?;
184        bpf_pin_object(self.fd.as_fd(), &path_string).map_err(|(_, io_error)| SyscallError {
185            call: "BPF_OBJ_PIN",
186            io_error,
187        })?;
188        Ok(PinnedLink::new(path.into(), self))
189    }
190}

200        // detach is a noop since it consumes self. once self is consumed, drop will be triggered
201        // and the link will be detached.
202        //
203        // Other links don't need to do this since they use define_link_wrapper!, but FdLink is a
204        // bit special in that it defines a custom ::new() so it can't use the macro.
205        Ok(())
206    }

215/// A pinned file descriptor link.
216///
217/// This link has been pinned to the BPF filesystem. On drop, the file descriptor that backs
218/// this link will be closed. Whether or not the program remains attached is dependent
219/// on the presence of the file in BPFFS.
220#[derive(Debug)]
221pub struct PinnedLink {
222    inner: FdLink,
223    path: PathBuf,
224}
225
226impl PinnedLink {
227    fn new(path: PathBuf, link: FdLink) -> Self {
228        Self { inner: link, path }
229    }
230
231    /// Creates a [`crate::programs::links::PinnedLink`] from a valid path on bpffs.
232    pub fn from_pin<P: AsRef<Path>>(path: P) -> Result<Self, LinkError> {
233        use std::os::unix::ffi::OsStrExt as _;
234
235        // TODO: avoid this unwrap by adding a new error variant.
236        let path_string = CString::new(path.as_ref().as_os_str().as_bytes()).unwrap();
237        let fd = bpf_get_object(&path_string).map_err(|(_, io_error)| {
238            LinkError::SyscallError(SyscallError {
239                call: "BPF_OBJ_GET",
240                io_error,
241            })
242        })?;
243        Ok(Self::new(path.as_ref().to_path_buf(), FdLink::new(fd)))
244    }
245
246    /// Removes the pinned link from the filesystem and returns an [`FdLink`].
247    pub fn unpin(self) -> Result<FdLink, io::Error> {
248        std::fs::remove_file(self.path)?;
249        Ok(self.inner)
250    }
251}

257/// The Link type used by programs that are attached with `bpf_prog_attach`.
258#[derive(Debug)]
259pub struct ProgAttachLink {
260    prog_fd: ProgramFd,
261    target_fd: crate::MockableFd,
262    attach_type: bpf_attach_type,
263}
264
265impl ProgAttachLink {
266    pub(crate) fn new(
267        prog_fd: ProgramFd,
268        target_fd: crate::MockableFd,
269        attach_type: bpf_attach_type,
270    ) -> Self {
271        Self {
272            prog_fd,
273            target_fd,
274            attach_type,
275        }
276    }
277
278    pub(crate) fn attach(
279        prog_fd: BorrowedFd<'_>,
280        target_fd: BorrowedFd<'_>,
281        attach_type: bpf_attach_type,
282        mode: CgroupAttachMode,
283    ) -> Result<Self, ProgramError> {
284        // The link is going to own this new file descriptor so we are
285        // going to need a duplicate whose lifetime we manage. Let's
286        // duplicate it prior to attaching it so the new file
287        // descriptor is closed at drop in case it fails to attach.
288        let prog_fd = prog_fd.try_clone_to_owned()?;
289        let prog_fd = crate::MockableFd::from_fd(prog_fd);
290        let target_fd = target_fd.try_clone_to_owned()?;
291        let target_fd = crate::MockableFd::from_fd(target_fd);
292        bpf_prog_attach(prog_fd.as_fd(), target_fd.as_fd(), attach_type, mode.into())?;
293
294        let prog_fd = ProgramFd(prog_fd);
295        Ok(Self {
296            prog_fd,
297            target_fd,
298            attach_type,
299        })
300    }
301}
302
303impl Link for ProgAttachLink {
304    type Id = ProgAttachLinkId;
305
306    fn id(&self) -> Self::Id {
307        ProgAttachLinkId(
308            self.prog_fd.as_fd().as_raw_fd(),
309            self.target_fd.as_raw_fd(),
310            self.attach_type,
311        )
312    }
313
314    fn detach(self) -> Result<(), ProgramError> {
315        bpf_prog_detach(
316            self.prog_fd.as_fd(),
317            self.target_fd.as_fd(),
318            self.attach_type,
319        )
320        .map_err(Into::into)
321    }
322}

324macro_rules! define_link_wrapper {
325    (#[$doc1:meta] $wrapper:ident, #[$doc2:meta] $wrapper_id:ident, $base:ident, $base_id:ident) => {
326        #[$doc2]
327        #[derive(Debug, Hash, Eq, PartialEq)]
328        pub struct $wrapper_id($base_id);
329
330        #[$doc1]
331        #[derive(Debug)]
332        pub struct $wrapper(Option<$base>);
333
334        #[allow(dead_code)]
335        // allow dead code since currently XDP/TC are the only consumers of inner and
336        // into_inner
337        impl $wrapper {
338            fn new(base: $base) -> $wrapper {
339                $wrapper(Some(base))
340            }
341
342            fn inner(&self) -> &$base {
343                self.0.as_ref().unwrap()
344            }
345
346            fn into_inner(mut self) -> $base {
347                self.0.take().unwrap()
348            }
349        }
350
351        impl Drop for $wrapper {
352            fn drop(&mut self) {
353                use crate::programs::links::Link;
354
355                if let Some(base) = self.0.take() {
356                    let _ = base.detach();
357                }
358            }
359        }
360
361        impl $crate::programs::Link for $wrapper {
362            type Id = $wrapper_id;
363
364            fn id(&self) -> Self::Id {
365                $wrapper_id(self.0.as_ref().unwrap().id())
366            }
367
368            fn detach(mut self) -> Result<(), ProgramError> {
369                self.0.take().unwrap().detach()
370            }
371        }
372
373        impl From<$base> for $wrapper {
374            fn from(b: $base) -> $wrapper {
375                $wrapper(Some(b))
376            }
377        }
378
379        impl From<$wrapper> for $base {
380            fn from(mut w: $wrapper) -> $base {
381                w.0.take().unwrap()
382            }
383        }
384    };
385}

418/// Arguments required for interacting with the kernel's multi-prog API.
419///
420/// # Minimum kernel version
421///
422/// The minimum kernel version required to use this feature is 6.6.0.
423///
424/// # Example
425///
426/// ```no_run
427/// # let mut bpf = aya::Ebpf::load(&[])?;
428/// use aya::programs::{tc, SchedClassifier, TcAttachType, tc::TcAttachOptions, LinkOrder};
429///
430/// let prog: &mut SchedClassifier = bpf.program_mut("redirect_ingress").unwrap().try_into()?;
431/// prog.load()?;
432/// let options = TcAttachOptions::TcxOrder(LinkOrder::first());
433/// prog.attach_with_options("eth0", TcAttachType::Ingress, options)?;
434///
435/// # Ok::<(), aya::EbpfError>(())
436/// ```
437#[derive(Debug)]
438pub struct LinkOrder {
439    pub(crate) link_ref: LinkRef,
440    pub(crate) flags: MprogFlags,
441}
442
443/// Ensure that default link ordering is to be attached last.
444impl Default for LinkOrder {
445    fn default() -> Self {
446        Self {
447            link_ref: LinkRef::Fd(0),
448            flags: MprogFlags::AFTER,
449        }
450    }
451}
452
453impl LinkOrder {
454    /// Attach before all other links.
455    pub fn first() -> Self {
456        Self {
457            link_ref: LinkRef::Id(0),
458            flags: MprogFlags::BEFORE,
459        }
460    }
461
462    /// Attach after all other links.
463    pub fn last() -> Self {
464        Self {
465            link_ref: LinkRef::Id(0),
466            flags: MprogFlags::AFTER,
467        }
468    }
469
470    /// Attach before the given link.
471    pub fn before_link<L: MultiProgLink>(link: &L) -> Result<Self, LinkError> {
472        Ok(Self {
473            link_ref: LinkRef::Fd(link.fd()?.as_raw_fd()),
474            flags: MprogFlags::BEFORE | MprogFlags::LINK,
475        })
476    }
477
478    /// Attach after the given link.
479    pub fn after_link<L: MultiProgLink>(link: &L) -> Result<Self, LinkError> {
480        Ok(Self {
481            link_ref: LinkRef::Fd(link.fd()?.as_raw_fd()),
482            flags: MprogFlags::AFTER | MprogFlags::LINK,
483        })
484    }
485
486    /// Attach before the given program.
487    pub fn before_program<P: MultiProgram>(program: &P) -> Result<Self, ProgramError> {
488        Ok(Self {
489            link_ref: LinkRef::Fd(program.fd()?.as_raw_fd()),
490            flags: MprogFlags::BEFORE,
491        })
492    }
493
494    /// Attach after the given program.
495    pub fn after_program<P: MultiProgram>(program: &P) -> Result<Self, ProgramError> {
496        Ok(Self {
497            link_ref: LinkRef::Fd(program.fd()?.as_raw_fd()),
498            flags: MprogFlags::AFTER,
499        })
500    }
501
502    /// Attach before the program with the given id.
503    pub fn before_program_id(id: ProgramId) -> Self {
504        Self {
505            link_ref: LinkRef::Id(id.0),
506            flags: MprogFlags::BEFORE | MprogFlags::ID,
507        }
508    }
509
510    /// Attach after the program with the given id.
511    pub fn after_program_id(id: ProgramId) -> Self {
512        Self {
513            link_ref: LinkRef::Id(id.0),
514            flags: MprogFlags::AFTER | MprogFlags::ID,
515        }
516    }
517}
