23/// A kernel probe.
24///
25/// Kernel probes are eBPF programs that can be attached to almost any function inside
26/// the kernel. They can be of two kinds:
27///
28/// - `kprobe`: get attached to the *start* of the target functions
29/// - `kretprobe`: get attached to the *return address* of the target functions
30///
31/// # Minimum kernel version
32///
33/// The minimum kernel version required to use this feature is 4.1.
34///
35/// # Examples
36///
37/// ```no_run
38/// # let mut bpf = Ebpf::load_file("ebpf_programs.o")?;
39/// use aya::{Ebpf, programs::KProbe};
40///
41/// let program: &mut KProbe = bpf.program_mut("intercept_wakeups").unwrap().try_into()?;
42/// program.load()?;
43/// program.attach("try_to_wake_up", 0)?;
44/// # Ok::<(), aya::EbpfError>(())
45/// ```
46#[derive(Debug)]
47#[doc(alias = "BPF_PROG_TYPE_KPROBE")]
48pub struct KProbe {
49    pub(crate) data: ProgramData<KProbeLink>,
50    pub(crate) kind: ProbeKind,
51}
52
53impl KProbe {
54    /// Loads the program inside the kernel.
55    pub fn load(&mut self) -> Result<(), ProgramError> {
56        load_program(BPF_PROG_TYPE_KPROBE, &mut self.data)
57    }
58
59    /// Returns `KProbe` if the program is a `kprobe`, or `KRetProbe` if the
60    /// program is a `kretprobe`.
61    pub fn kind(&self) -> ProbeKind {
62        self.kind
63    }
64
65    /// Attaches the program.
66    ///
67    /// Attaches the probe to the given function name inside the kernel. If
68    /// `offset` is non-zero, it is added to the address of the target
69    /// function.
70    ///
71    /// If the program is a `kprobe`, it is attached to the *start* address of the target function.
72    /// Conversely if the program is a `kretprobe`, it is attached to the return address of the
73    /// target function.
74    ///
75    /// The returned value can be used to detach from the given function, see [KProbe::detach].
76    pub fn attach<T: AsRef<OsStr>>(
77        &mut self,
78        fn_name: T,
79        offset: u64,
80    ) -> Result<KProbeLinkId, ProgramError> {
81        attach(&mut self.data, self.kind, fn_name.as_ref(), offset, None)
82    }
83
84    /// Detaches the program.
85    ///
86    /// See [KProbe::attach].
87    pub fn detach(&mut self, link_id: KProbeLinkId) -> Result<(), ProgramError> {
88        self.data.links.remove(link_id)
89    }
90
91    /// Takes ownership of the link referenced by the provided link_id.
92    ///
93    /// The link will be detached on `Drop` and the caller is now responsible
94    /// for managing its lifetime.
95    pub fn take_link(&mut self, link_id: KProbeLinkId) -> Result<KProbeLink, ProgramError> {
96        self.data.take_link(link_id)
97    }
98
99    /// Creates a program from a pinned entry on a bpffs.
100    ///
101    /// Existing links will not be populated. To work with existing links you should use [`crate::programs::links::PinnedLink`].
102    ///
103    /// On drop, any managed links are detached and the program is unloaded. This will not result in
104    /// the program being unloaded from the kernel if it is still pinned.
105    pub fn from_pin<P: AsRef<Path>>(path: P, kind: ProbeKind) -> Result<Self, ProgramError> {
106        let data = ProgramData::from_pinned_path(path, VerifierLogLevel::default())?;
107        Ok(Self { data, kind })
108    }
109}

120/// The type returned when attaching a [`KProbe`] fails.
121#[derive(Debug, Error)]
122pub enum KProbeError {
123    /// Error detaching from debugfs
124    #[error("`{filename}`")]
125    FileError {
126        /// The file name
127        filename: PathBuf,
128        /// The [`io::Error`] returned from the file operation
129        #[source]
130        io_error: io::Error,
131    },
132}

134impl TryFrom<KProbeLink> for FdLink {
135    type Error = LinkError;
136
137    fn try_from(value: KProbeLink) -> Result<Self, Self::Error> {
138        if let PerfLinkInner::FdLink(fd) = value.into_inner() {
139            Ok(fd)
140        } else {
141            Err(LinkError::InvalidLink)
142        }
143    }
144}
145
146impl TryFrom<FdLink> for KProbeLink {
147    type Error = LinkError;
148
149    fn try_from(fd_link: FdLink) -> Result<Self, Self::Error> {
150        let info = bpf_link_get_info_by_fd(fd_link.fd.as_fd())?;
151        if info.type_ == (bpf_link_type::BPF_LINK_TYPE_KPROBE_MULTI as u32) {
152            return Ok(Self::new(PerfLinkInner::FdLink(fd_link)));
153        }
154        Err(LinkError::InvalidLink)
155    }
156}
