15/// The type returned when attaching a [`SocketFilter`] fails.
16#[derive(Debug, Error)]
17pub enum SocketFilterError {
18    /// Setting the `SO_ATTACH_BPF` socket option failed.
19    #[error("setsockopt SO_ATTACH_BPF failed")]
20    SoAttachEbpfError {
21        /// original [`io::Error`]
22        #[source]
23        io_error: io::Error,
24    },

27/// A program used to inspect and filter incoming packets on a socket.
28///
29/// [`SocketFilter`] programs are attached on sockets and can be used to inspect
30/// and filter incoming packets.
31///
32/// # Minimum kernel version
33///
34/// The minimum kernel version required to use this feature is 4.0.
35///
36/// # Examples
37///
38/// ```no_run
39/// # #[derive(Debug, thiserror::Error)]
40/// # enum Error {
41/// #     #[error(transparent)]
42/// #     IO(#[from] std::io::Error),
43/// #     #[error(transparent)]
44/// #     Map(#[from] aya::maps::MapError),
45/// #     #[error(transparent)]
46/// #     Program(#[from] aya::programs::ProgramError),
47/// #     #[error(transparent)]
48/// #     Ebpf(#[from] aya::EbpfError)
49/// # }
50/// # let mut bpf = aya::Ebpf::load(&[])?;
51/// use std::net::TcpStream;
52/// use aya::programs::SocketFilter;
53///
54/// let mut client = TcpStream::connect("127.0.0.1:1234")?;
55/// let prog: &mut SocketFilter = bpf.program_mut("filter_packets").unwrap().try_into()?;
56/// prog.load()?;
57/// prog.attach(&client)?;
58/// # Ok::<(), Error>(())
59/// ```
60#[derive(Debug)]
61#[doc(alias = "BPF_PROG_TYPE_SOCKET_FILTER")]
62pub struct SocketFilter {
63    pub(crate) data: ProgramData<SocketFilterLink>,
64}
65
66impl SocketFilter {
67    /// Loads the program inside the kernel.
68    pub fn load(&mut self) -> Result<(), ProgramError> {
69        load_program(BPF_PROG_TYPE_SOCKET_FILTER, &mut self.data)
70    }
71
72    /// Attaches the filter on the given socket.
73    ///
74    /// The returned value can be used to detach from the socket, see [SocketFilter::detach].
75    pub fn attach<T: AsFd>(&mut self, socket: T) -> Result<SocketFilterLinkId, ProgramError> {
76        let prog_fd = self.fd()?;
77        let prog_fd = prog_fd.as_fd();
78        let prog_fd = prog_fd.as_raw_fd();
79        let socket = socket.as_fd();
80        let socket = socket.as_raw_fd();
81
82        let ret = unsafe {
83            setsockopt(
84                socket,
85                SOL_SOCKET,
86                SO_ATTACH_BPF as i32,
87                &prog_fd as *const _ as *const _,
88                mem::size_of::<RawFd>() as u32,
89            )
90        };
91        if ret < 0 {
92            return Err(SocketFilterError::SoAttachEbpfError {
93                io_error: io::Error::last_os_error(),
94            }
95            .into());
96        }
97
98        self.data.links.insert(SocketFilterLink { socket, prog_fd })
99    }
100
101    /// Detaches the program.
102    ///
103    /// See [SocketFilter::attach].
104    pub fn detach(&mut self, link_id: SocketFilterLinkId) -> Result<(), ProgramError> {
105        self.data.links.remove(link_id)
106    }
107
108    /// Takes ownership of the link referenced by the provided link_id.
109    ///
110    /// The link will be detached on `Drop` and the caller is now responsible
111    /// for managing its lifetime.
112    pub fn take_link(
113        &mut self,
114        link_id: SocketFilterLinkId,
115    ) -> Result<SocketFilterLink, ProgramError> {
116        self.data.take_link(link_id)
117    }
118}
119
120/// The type returned by [SocketFilter::attach]. Can be passed to [SocketFilter::detach].
121#[derive(Debug, Hash, Eq, PartialEq)]
122pub struct SocketFilterLinkId(RawFd, RawFd);
123
124/// A SocketFilter Link
125#[derive(Debug)]
126pub struct SocketFilterLink {
127    socket: RawFd,
128    prog_fd: RawFd,
129}
130
131impl Link for SocketFilterLink {
132    type Id = SocketFilterLinkId;
133
134    fn id(&self) -> Self::Id {
135        SocketFilterLinkId(self.socket, self.prog_fd)
136    }
137
138    fn detach(self) -> Result<(), ProgramError> {
139        unsafe {
140            setsockopt(
141                self.socket,
142                SOL_SOCKET,
143                SO_DETACH_BPF as i32,
144                &self.prog_fd as *const _ as *const _,
145                mem::size_of::<RawFd>() as u32,
146            );
147        }
148        Ok(())
149    }
150}
