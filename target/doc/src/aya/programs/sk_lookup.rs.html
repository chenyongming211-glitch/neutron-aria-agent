11/// A program used to redirect incoming packets to a local socket.
12///
13/// [`SkLookup`] programs are attached to network namespaces to provide programmable
14/// socket lookup for TCP/UDP when a packet is to be delievered locally.
15///
16/// You may attach multiple programs to the same namespace and they are executed
17/// in the order they were attached.
18///
19/// # Minimum kernel version
20///
21/// The minimum kernel version required to use this feature is 5.9.
22///
23/// # Examples
24///
25/// ```no_run
26/// # #[derive(Debug, thiserror::Error)]
27/// # enum Error {
28/// #     #[error(transparent)]
29/// #     IO(#[from] std::io::Error),
30/// #     #[error(transparent)]
31/// #     Map(#[from] aya::maps::MapError),
32/// #     #[error(transparent)]
33/// #     Program(#[from] aya::programs::ProgramError),
34/// #     #[error(transparent)]
35/// #     Ebpf(#[from] aya::EbpfError)
36/// # }
37/// # let mut bpf = aya::Ebpf::load(&[])?;
38/// use std::fs::File;
39/// use aya::programs::SkLookup;
40///
41/// let file = File::open("/var/run/netns/test")?;
42/// let program: &mut SkLookup = bpf.program_mut("sk_lookup").unwrap().try_into()?;
43/// program.load()?;
44/// program.attach(file)?;
45/// # Ok::<(), Error>(())
46/// ```
47#[derive(Debug)]
48#[doc(alias = "BPF_PROG_TYPE_SK_LOOKUP")]
49pub struct SkLookup {
50    pub(crate) data: ProgramData<SkLookupLink>,
51}
52
53impl SkLookup {
54    /// Loads the program inside the kernel.
55    pub fn load(&mut self) -> Result<(), ProgramError> {
56        self.data.expected_attach_type = Some(BPF_SK_LOOKUP);
57        load_program(BPF_PROG_TYPE_SK_LOOKUP, &mut self.data)
58    }
59
60    /// Attaches the program to the given network namespace.
61    ///
62    /// The returned value can be used to detach, see [SkLookup::detach].
63    pub fn attach<T: AsFd>(&mut self, netns: T) -> Result<SkLookupLinkId, ProgramError> {
64        let prog_fd = self.fd()?;
65        let prog_fd = prog_fd.as_fd();
66        let netns_fd = netns.as_fd();
67
68        let link_fd = bpf_link_create(
69            prog_fd,
70            LinkTarget::Fd(netns_fd),
71            BPF_SK_LOOKUP,
72            None,
73            0,
74            None,
75        )
76        .map_err(|(_, io_error)| SyscallError {
77            call: "bpf_link_create",
78            io_error,
79        })?;
80        self.data
81            .links
82            .insert(SkLookupLink::new(FdLink::new(link_fd)))
83    }
84
85    /// Takes ownership of the link referenced by the provided link_id.
86    ///
87    /// The link will be detached on `Drop` and the caller is now responsible
88    /// for managing its lifetime.
89    pub fn take_link(&mut self, link_id: SkLookupLinkId) -> Result<SkLookupLink, ProgramError> {
90        self.data.take_link(link_id)
91    }
92
93    /// Detaches the program.
94    ///
95    /// See [SkLookup::attach].
96    pub fn detach(&mut self, link_id: SkLookupLinkId) -> Result<(), ProgramError> {
97        self.data.links.remove(link_id)
98    }
99}
