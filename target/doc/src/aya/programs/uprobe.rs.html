37/// An user space probe.
38///
39/// User probes are eBPF programs that can be attached to any userspace
40/// function. They can be of two kinds:
41///
42/// - `uprobe`: get attached to the *start* of the target functions
43/// - `uretprobe`: get attached to the *return address* of the target functions
44#[derive(Debug)]
45#[doc(alias = "BPF_PROG_TYPE_KPROBE")]
46pub struct UProbe {
47    pub(crate) data: ProgramData<UProbeLink>,
48    pub(crate) kind: ProbeKind,
49}
50
51impl UProbe {
52    /// Loads the program inside the kernel.
53    pub fn load(&mut self) -> Result<(), ProgramError> {
54        load_program(BPF_PROG_TYPE_KPROBE, &mut self.data)
55    }
56
57    /// Returns `UProbe` if the program is a `uprobe`, or `URetProbe` if the
58    /// program is a `uretprobe`.
59    pub fn kind(&self) -> ProbeKind {
60        self.kind
61    }
62
63    /// Attaches the program.
64    ///
65    /// Attaches the uprobe to the function `fn_name` defined in the `target`.
66    /// If `offset` is non-zero, it is added to the address of the target
67    /// function. If `pid` is not `None`, the program executes only when the target
68    /// function is executed by the given `pid`.
69    ///
70    /// The `target` argument can be an absolute path to a binary or library, or
71    /// a library name (eg: `"libc"`).
72    ///
73    /// If the program is an `uprobe`, it is attached to the *start* address of the target
74    /// function.  Instead if the program is a `uretprobe`, it is attached to the return address of
75    /// the target function.
76    ///
77    /// The returned value can be used to detach, see [UProbe::detach].
78    pub fn attach<T: AsRef<Path>>(
79        &mut self,
80        fn_name: Option<&str>,
81        offset: u64,
82        target: T,
83        pid: Option<pid_t>,
84    ) -> Result<UProbeLinkId, ProgramError> {
85        let path = resolve_attach_path(target.as_ref(), pid)?;
86
87        let sym_offset = if let Some(fn_name) = fn_name {
88            resolve_symbol(&path, fn_name).map_err(|error| UProbeError::SymbolError {
89                symbol: fn_name.to_string(),
90                error: Box::new(error),
91            })?
92        } else {
93            0
94        };
95
96        let path = path.as_os_str();
97        attach(&mut self.data, self.kind, path, sym_offset + offset, pid)
98    }
99
100    /// Detaches the program.
101    ///
102    /// See [UProbe::attach].
103    pub fn detach(&mut self, link_id: UProbeLinkId) -> Result<(), ProgramError> {
104        self.data.links.remove(link_id)
105    }
106
107    /// Takes ownership of the link referenced by the provided link_id.
108    ///
109    /// The link will be detached on `Drop` and the caller is now responsible
110    /// for managing its lifetime.
111    pub fn take_link(&mut self, link_id: UProbeLinkId) -> Result<UProbeLink, ProgramError> {
112        self.data.take_link(link_id)
113    }
114
115    /// Creates a program from a pinned entry on a bpffs.
116    ///
117    /// Existing links will not be populated. To work with existing links you should use [`crate::programs::links::PinnedLink`].
118    ///
119    /// On drop, any managed links are detached and the program is unloaded. This will not result in
120    /// the program being unloaded from the kernel if it is still pinned.
121    pub fn from_pin<P: AsRef<Path>>(path: P, kind: ProbeKind) -> Result<Self, ProgramError> {
122        let data = ProgramData::from_pinned_path(path, VerifierLogLevel::default())?;
123        Ok(Self { data, kind })
124    }
125}

127fn resolve_attach_path(target: &Path, pid: Option<pid_t>) -> Result<Cow<'_, Path>, UProbeError> {
128    // Look up the path for the target. If it there is a pid, and the target is a library name
129    // that is in the process's memory map, use the path of that library. Otherwise, use the target as-is.
130    pid.and_then(|pid| {
131        find_lib_in_proc_maps(pid, target)
132            .map_err(|io_error| UProbeError::FileError {
133                filename: Path::new("/proc").join(pid.to_string()).join("maps"),
134                io_error,
135            })
136            .map(|v| v.map(Cow::Owned))
137            .transpose()
138    })
139    .or_else(|| target.is_absolute().then(|| Ok(Cow::Borrowed(target))))
140    .or_else(|| {
141        LD_SO_CACHE
142            .as_ref()
143            .map_err(|io_error| UProbeError::InvalidLdSoCache { io_error })
144            .map(|cache| cache.resolve(target).map(Cow::Borrowed))
145            .transpose()
146    })
147    .unwrap_or_else(|| {
148        Err(UProbeError::InvalidTarget {
149            path: target.to_owned(),
150        })
151    })
152}

197    type Error = LinkError;

208/// The type returned when attaching an [`UProbe`] fails.
209#[derive(Debug, Error)]
210pub enum UProbeError {
211    /// There was an error parsing `/etc/ld.so.cache`.
212    #[error("error reading `{}` file", LD_SO_CACHE_FILE)]
213    InvalidLdSoCache {
214        /// the original [`io::Error`]
215        #[source]
216        io_error: &'static io::Error,
217    },
218
219    /// The target program could not be found.
220    #[error("could not resolve uprobe target `{path}`")]
221    InvalidTarget {
222        /// path to target
223        path: PathBuf,
224    },
225
226    /// There was an error resolving the target symbol.
227    #[error("error resolving symbol")]
228    SymbolError {
229        /// symbol name
230        symbol: String,
231        /// the original error
232        #[source]
233        error: Box<dyn Error + Send + Sync>,
234    },
235
236    /// There was an error accessing `filename`.
237    #[error("`{filename}`")]
238    FileError {
239        /// The file name
240        filename: PathBuf,
241        /// The [`io::Error`] returned from the file operation
242        #[source]
243        io_error: io::Error,
244    },
245}

247fn proc_maps_libs(pid: pid_t) -> Result<Vec<(OsString, PathBuf)>, io::Error> {
248    use std::os::unix::ffi::OsStrExt as _;
249
250    let maps_file = format!("/proc/{pid}/maps");
251    let data = fs::read(maps_file)?;
252
253    let libs = data
254        .split(|b| b == &b'\n')
255        .filter_map(|mut line| {
256            while let [stripped @ .., c] = line {
257                if c.is_ascii_whitespace() {
258                    line = stripped;
259                    continue;
260                }
261                break;
262            }
263            let path = line.split(|b| b.is_ascii_whitespace()).last()?;
264            let path = Path::new(OsStr::from_bytes(path));
265            path.is_absolute()
266                .then(|| {
267                    path.file_name()
268                        .map(|file_name| (file_name.to_owned(), path.to_owned()))
269                })
270                .flatten()
271        })
272        .collect();
273    Ok(libs)
274}

276fn find_lib_in_proc_maps(pid: pid_t, lib: &Path) -> Result<Option<PathBuf>, io::Error> {
277    let libs = proc_maps_libs(pid)?;
278
279    let lib = lib.as_os_str();
280    let lib = lib.strip_suffix(OsStr::new(".so")).unwrap_or(lib);
281
282    Ok(libs.into_iter().find_map(|(file_name, path)| {
283        file_name.strip_prefix(lib).and_then(|suffix| {
284            (suffix.starts_with(OsStr::new(".so")) || suffix.starts_with(OsStr::new("-")))
285                .then_some(path)
286        })
287    }))
288}

293    value: OsString,
294    _flags: i32,

297#[derive(Debug)]
298pub(crate) struct LdSoCache {
299    entries: Vec<CacheEntry>,
300}
301
302impl LdSoCache {
303    fn load<T: AsRef<Path>>(path: T) -> Result<Self, io::Error> {
304        let data = fs::read(path)?;
305        Self::parse(&data)
306    }
307
308    fn parse(data: &[u8]) -> Result<Self, io::Error> {
309        let mut cursor = Cursor::new(data);
310
311        let read_u32 = |cursor: &mut Cursor<_>| -> Result<u32, io::Error> {
312            let mut buf = [0u8; mem::size_of::<u32>()];
313            cursor.read_exact(&mut buf)?;
314
315            Ok(u32::from_ne_bytes(buf))
316        };
317
318        let read_i32 = |cursor: &mut Cursor<_>| -> Result<i32, io::Error> {
319            let mut buf = [0u8; mem::size_of::<i32>()];
320            cursor.read_exact(&mut buf)?;
321
322            Ok(i32::from_ne_bytes(buf))
323        };
324
325        // Check for new format
326        let mut buf = [0u8; LD_SO_CACHE_HEADER_NEW.len()];
327        cursor.read_exact(&mut buf)?;
328        let header = std::str::from_utf8(&buf).map_err(|_| {
329            io::Error::new(io::ErrorKind::InvalidData, "invalid ld.so.cache header")
330        })?;
331
332        let new_format = header == LD_SO_CACHE_HEADER_NEW;
333
334        // Check for old format
335        if !new_format {
336            cursor.set_position(0);
337            let mut buf = [0u8; LD_SO_CACHE_HEADER_OLD.len()];
338            cursor.read_exact(&mut buf)?;
339            let header = std::str::from_utf8(&buf).map_err(|_| {
340                io::Error::new(io::ErrorKind::InvalidData, "invalid ld.so.cache header")
341            })?;
342
343            if header != LD_SO_CACHE_HEADER_OLD {
344                return Err(io::Error::new(
345                    io::ErrorKind::InvalidData,
346                    "invalid ld.so.cache header",
347                ));
348            }
349        }
350
351        let num_entries = read_u32(&mut cursor)?;
352
353        if new_format {
354            cursor.consume(6 * mem::size_of::<u32>());
355        }
356
357        let offset = if !new_format {
358            cursor.position() as usize + num_entries as usize * 12
359        } else {
360            0
361        };
362
363        let entries = (0..num_entries)
364            .map(|_: u32| {
365                let flags = read_i32(&mut cursor)?;
366                let k_pos = read_u32(&mut cursor)? as usize;
367                let v_pos = read_u32(&mut cursor)? as usize;
368
369                if new_format {
370                    cursor.consume(12);
371                }
372
373                let read_str = |pos| {
374                    use std::os::unix::ffi::OsStrExt as _;
375                    OsStr::from_bytes(
376                        unsafe {
377                            CStr::from_ptr(
378                                cursor.get_ref()[offset + pos..].as_ptr() as *const c_char
379                            )
380                        }
381                        .to_bytes(),
382                    )
383                    .to_owned()
384                };
385
386                let key = read_str(k_pos);
387                let value = read_str(v_pos);
388
389                Ok::<_, io::Error>(CacheEntry {
390                    key,
391                    value,
392                    _flags: flags,
393                })
394            })
395            .collect::<Result<_, _>>()?;
396
397        Ok(Self { entries })
398    }
399
400    fn resolve(&self, lib: &Path) -> Option<&Path> {
401        let lib = lib.as_os_str();
402        let lib = lib.strip_suffix(OsStr::new(".so")).unwrap_or(lib);
403        self.entries
404            .iter()
405            .find_map(|CacheEntry { key, value, _flags }| {
406                key.strip_prefix(lib).and_then(|suffix| {
407                    suffix
408                        .starts_with(OsStr::new(".so"))
409                        .then_some(Path::new(value.as_os_str()))
410                })
411            })
412    }
413}

415#[derive(Error, Debug)]
416enum ResolveSymbolError {
417    #[error(transparent)]
418    Io(#[from] io::Error),
419
420    #[error("error parsing ELF")]
421    Object(#[from] object::Error),
422
423    #[error("unknown symbol `{0}`")]
424    Unknown(String),
425
426    #[error("symbol `{0}` does not appear in section")]
427    NotInSection(String),
428
429    #[error("symbol `{0}` in section `{1:?}` which has no offset")]
430    SectionFileRangeNone(String, Result<String, object::Error>),
431
432    #[error("failed to access debuglink file `{0}`: `{1}`")]
433    DebuglinkAccessError(String, io::Error),
434
435    #[error("symbol `{0}` not found, mismatched build IDs in main and debug files")]
436    BuildIdMismatch(String),
437}

439fn construct_debuglink_path(
440    filename: &[u8],
441    main_path: &Path,
442) -> Result<PathBuf, ResolveSymbolError> {
443    let filename_str = OsStr::from_bytes(filename);
444    let debuglink_path = Path::new(filename_str);
445
446    let resolved_path = if debuglink_path.is_relative() {
447        // If the debug path is relative, resolve it against the parent of the main path
448        main_path.parent().map_or_else(
449            || PathBuf::from(debuglink_path), // Use original if no parent
450            |parent| parent.join(debuglink_path),
451        )
452    } else {
453        // If the path is not relative, just use original
454        PathBuf::from(debuglink_path)
455    };
456
457    Ok(resolved_path)
458}

460fn verify_build_ids<'a>(
461    main_obj: &'a object::File<'a>,
462    debug_obj: &'a object::File<'a>,
463    symbol_name: &str,
464) -> Result<(), ResolveSymbolError> {
465    let main_build_id = main_obj.build_id().ok().flatten();
466    let debug_build_id = debug_obj.build_id().ok().flatten();
467
468    match (debug_build_id, main_build_id) {
469        (Some(debug_build_id), Some(main_build_id)) => {
470            // Only perform a comparison if both build IDs are present
471            if debug_build_id != main_build_id {
472                return Err(ResolveSymbolError::BuildIdMismatch(symbol_name.to_owned()));
473            }
474            Ok(())
475        }
476        _ => Ok(()),
477    }
478}

480fn find_debug_path_in_object<'a>(
481    obj: &'a object::File<'a>,
482    main_path: &Path,
483    symbol: &str,
484) -> Result<PathBuf, ResolveSymbolError> {
485    match obj.gnu_debuglink() {
486        Ok(Some((filename, _))) => construct_debuglink_path(filename, main_path),
487        Ok(None) => Err(ResolveSymbolError::Unknown(symbol.to_string())),
488        Err(err) => Err(ResolveSymbolError::Object(err)),
489    }
490}

492fn find_symbol_in_object<'a>(obj: &'a object::File<'a>, symbol: &str) -> Option<Symbol<'a, 'a>> {
493    obj.dynamic_symbols()
494        .chain(obj.symbols())
495        .find(|sym| sym.name().map(|name| name == symbol).unwrap_or(false))
496}

498fn resolve_symbol(path: &Path, symbol: &str) -> Result<u64, ResolveSymbolError> {
499    let data = fs::read(path)?;
500    let obj = object::read::File::parse(&*data)?;
501
502    let mut debug_data = Vec::default();
503    let mut debug_obj_keeper = None;
504
505    let sym = find_symbol_in_object(&obj, symbol).map_or_else(
506        || {
507            // Only search in the debug object if the symbol was not found in the main object
508            let debug_path = find_debug_path_in_object(&obj, path, symbol)?;
509            debug_data = fs::read(&debug_path).map_err(|e| {
510                ResolveSymbolError::DebuglinkAccessError(
511                    debug_path
512                        .to_str()
513                        .unwrap_or("Debuglink path missing")
514                        .to_string(),
515                    e,
516                )
517            })?;
518            let debug_obj = object::read::File::parse(&*debug_data)?;
519
520            verify_build_ids(&obj, &debug_obj, symbol)?;
521
522            debug_obj_keeper = Some(debug_obj);
523            find_symbol_in_object(debug_obj_keeper.as_ref().unwrap(), symbol)
524                .ok_or_else(|| ResolveSymbolError::Unknown(symbol.to_string()))
525        },
526        Ok,
527    )?;
528
529    let needs_addr_translation = matches!(
530        obj.kind(),
531        object::ObjectKind::Dynamic | object::ObjectKind::Executable
532    );
533    if !needs_addr_translation {
534        Ok(sym.address())
535    } else {
536        let index = sym
537            .section_index()
538            .ok_or_else(|| ResolveSymbolError::NotInSection(symbol.to_string()))?;
539        let section = obj.section_by_index(index)?;
540        let (offset, _size) = section.file_range().ok_or_else(|| {
541            ResolveSymbolError::SectionFileRangeNone(
542                symbol.to_string(),
543                section.name().map(str::to_owned),
544            )
545        })?;
546        Ok(sym.address() - section.address() + offset)
547    }
548}

647    fn aligned_slice(vec: &mut Vec<u8>) -> &mut [u8] {
648        let alignment = 8;
649
650        let original_size = vec.len();
651        let total_size = original_size + alignment - 1;
652
653        if vec.capacity() < total_size {
654            vec.reserve(total_size - vec.capacity());
655        }
656
657        if vec.len() < total_size {
658            vec.resize(total_size, 0);
659        }
660
661        let ptr = vec.as_ptr() as usize;
662
663        let aligned_ptr = (ptr + alignment - 1) & !(alignment - 1);
664
665        let offset = aligned_ptr - ptr;
666
667        if offset > 0 {
668            let tmp = vec.len();
669            vec.copy_within(0..tmp - offset, offset);
670        }
671
672        &mut vec[offset..offset + original_size]
673    }
