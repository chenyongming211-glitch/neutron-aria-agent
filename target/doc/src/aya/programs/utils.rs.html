17/// Attaches the program to a raw tracepoint.
18pub(crate) fn attach_raw_tracepoint<T: Link + From<FdLink>>(
19    program_data: &mut ProgramData<T>,
20    tp_name: Option<&CStr>,
21) -> Result<T::Id, ProgramError> {
22    let prog_fd = program_data.fd()?;
23    let prog_fd = prog_fd.as_fd();
24    let pfd =
25        bpf_raw_tracepoint_open(tp_name, prog_fd).map_err(|(_code, io_error)| SyscallError {
26            call: "bpf_raw_tracepoint_open",
27            io_error,
28        })?;
29
30    program_data.links.insert(FdLink::new(pfd).into())
31}

33/// Find tracefs filesystem path.
34pub(crate) fn find_tracefs_path() -> Result<&'static Path, ProgramError> {
35    static TRACE_FS: LazyLock<Option<&'static Path>> = LazyLock::new(|| {
36        [
37            Path::new("/sys/kernel/tracing"),
38            Path::new("/sys/kernel/debug/tracing"),
39        ]
40        .into_iter()
41        .find(|&mount| {
42            // Check that the mount point exists and is not empty
43            // Documented here: (https://www.kernel.org/doc/Documentation/trace/ftrace.txt)
44            // In some cases, tracefs will only mount at /sys/kernel/debug/tracing
45            // but, the kernel will still create the directory /sys/kernel/tracing.
46            // The user may be expected to manually mount the directory in order for it to
47            // exist in /sys/kernel/tracing according to the documentation.
48            mount.exists()
49                && match mount.read_dir() {
50                    Ok(mut entries) => entries.next().is_some(),
51                    Err(io::Error { .. }) => false,
52                }
53        })
54    });
55
56    TRACE_FS
57        .as_deref()
58        .ok_or_else(|| io::Error::new(io::ErrorKind::Other, "tracefs not found").into())
59}

61/// The time at which the system is booted.
62pub(crate) fn boot_time() -> SystemTime {
63    let get_time = |clock_id| {
64        let mut time = unsafe { std::mem::zeroed::<libc::timespec>() };
65        assert_eq!(
66            unsafe { libc::clock_gettime(clock_id, &mut time) },
67            0,
68            "clock_gettime({}, _)",
69            clock_id
70        );
71        let libc::timespec { tv_sec, tv_nsec } = time;
72
73        Duration::new(tv_sec as u64, tv_nsec as u32)
74    };
75    let since_boot = get_time(libc::CLOCK_BOOTTIME);
76    let since_epoch = get_time(libc::CLOCK_REALTIME);
77    UNIX_EPOCH + since_epoch - since_boot
78}

80/// Get the specified information from a file descriptor's fdinfo.
81pub(crate) fn get_fdinfo(fd: BorrowedFd<'_>, key: &str) -> Result<u32, ProgramError> {
82    let info = File::open(format!("/proc/self/fdinfo/{}", fd.as_raw_fd()))?;
83    let reader = BufReader::new(info);
84    for line in reader.lines() {
85        let line = line?;
86        if !line.contains(key) {
87            continue;
88        }
89
90        let (_key, val) = line.rsplit_once('\t').unwrap();
91
92        return Ok(val.parse().unwrap());
93    }
94
95    Ok(0)
96}
