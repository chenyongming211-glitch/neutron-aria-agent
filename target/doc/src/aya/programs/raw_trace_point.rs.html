12/// A program that can be attached at a pre-defined kernel trace point.
13///
14/// Unlike [`TracePoint`](super::TracePoint), the kernel does not pre-process
15/// the arguments before calling the program.
16///
17/// The kernel provides a set of pre-defined trace points that eBPF programs can
18/// be attached to. See`/sys/kernel/debug/tracing/events` for a list of which
19/// events can be traced.
20///
21/// # Minimum kernel version
22///
23/// The minimum kernel version required to use this feature is 4.17.
24///
25/// # Examples
26///
27/// ```no_run
28/// # let mut bpf = aya::Ebpf::load(&[])?;
29/// use aya::programs::RawTracePoint;
30///
31/// let program: &mut RawTracePoint = bpf.program_mut("sys_enter").unwrap().try_into()?;
32/// program.load()?;
33/// program.attach("sys_enter")?;
34/// # Ok::<(), aya::EbpfError>(())
35/// ```
36#[derive(Debug)]
37#[doc(alias = "BPF_PROG_TYPE_RAW_TRACEPOINT")]
38pub struct RawTracePoint {
39    pub(crate) data: ProgramData<RawTracePointLink>,
40}
41
42impl RawTracePoint {
43    /// Loads the program inside the kernel.
44    pub fn load(&mut self) -> Result<(), ProgramError> {
45        load_program(BPF_PROG_TYPE_RAW_TRACEPOINT, &mut self.data)
46    }
47
48    /// Attaches the program to the given tracepoint.
49    ///
50    /// The returned value can be used to detach, see [RawTracePoint::detach].
51    pub fn attach(&mut self, tp_name: &str) -> Result<RawTracePointLinkId, ProgramError> {
52        let tp_name_c = CString::new(tp_name).unwrap();
53        attach_raw_tracepoint(&mut self.data, Some(&tp_name_c))
54    }
55
56    /// Detaches from a tracepoint.
57    ///
58    /// See [RawTracePoint::attach].
59    pub fn detach(&mut self, link_id: RawTracePointLinkId) -> Result<(), ProgramError> {
60        self.data.links.remove(link_id)
61    }
62
63    /// Takes ownership of the link referenced by the provided link_id.
64    ///
65    /// The link will be detached on `Drop` and the caller is now responsible
66    /// for managing its lifetime.
67    pub fn take_link(
68        &mut self,
69        link_id: RawTracePointLinkId,
70    ) -> Result<RawTracePointLink, ProgramError> {
71        self.data.take_link(link_id)
72    }
73}

76    /// The link used by [RawTracePoint] programs.
77    RawTracePointLink,
78    /// The type returned by [RawTracePoint::attach]. Can be passed to [RawTracePoint::detach].
79    RawTracePointLinkId,
80    FdLink,
81    FdLinkId
