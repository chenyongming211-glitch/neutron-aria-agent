18/// The type returned when loading or attaching an [`Extension`] fails.
19#[derive(Debug, Error)]
20pub enum ExtensionError {
21    /// Target BPF program does not have BTF loaded to the kernel.
22    #[error("target BPF program does not have BTF loaded to the kernel")]
23    NoBTF,
24}
25
26/// A program used to extend existing BPF programs.
27///
28/// [`Extension`] programs can be loaded to replace a global
29/// function in a program that has already been loaded.
30///
31/// # Minimum kernel version
32///
33/// The minimum kernel version required to use this feature is 5.9
34///
35/// # Examples
36///
37/// ```no_run
38/// use aya::{EbpfLoader, programs::{Xdp, XdpFlags, Extension}};
39///
40/// let mut bpf = EbpfLoader::new().extension("extension").load_file("app.o")?;
41/// let prog: &mut Xdp = bpf.program_mut("main").unwrap().try_into()?;
42/// prog.load()?;
43/// prog.attach("eth0", XdpFlags::default())?;
44///
45/// let prog_fd = prog.fd().unwrap();
46/// let prog_fd = prog_fd.try_clone().unwrap();
47/// let ext: &mut Extension = bpf.program_mut("extension").unwrap().try_into()?;
48/// ext.load(prog_fd, "function_to_replace")?;
49/// ext.attach()?;
50/// Ok::<(), aya::EbpfError>(())
51/// ```
52#[derive(Debug)]
53#[doc(alias = "BPF_PROG_TYPE_EXT")]
54pub struct Extension {
55    pub(crate) data: ProgramData<ExtensionLink>,
56}
57
58impl Extension {
59    /// Loads the extension inside the kernel.
60    ///
61    /// Prepares the code included in the extension to replace the code of the function
62    /// `func_name` within the eBPF program represented by the `program` file descriptor.
63    /// This requires that both the [`Extension`] and `program` have had their BTF
64    /// loaded into the kernel.
65    ///
66    /// The BPF verifier requires that we specify the target program and function name
67    /// at load time, so it can identify that the program and target are BTF compatible
68    /// and to enforce this constraint when programs are attached.
69    ///
70    /// The extension code will be loaded but inactive until it's attached.
71    /// There are no restrictions on what functions may be replaced, so you could replace
72    /// the main entry point of your program with an extension.
73    pub fn load(&mut self, program: ProgramFd, func_name: &str) -> Result<(), ProgramError> {
74        let (btf_fd, btf_id) = get_btf_info(program.as_fd(), func_name)?;
75
76        self.data.attach_btf_obj_fd = Some(btf_fd);
77        self.data.attach_prog_fd = Some(program);
78        self.data.attach_btf_id = Some(btf_id);
79        load_program(BPF_PROG_TYPE_EXT, &mut self.data)
80    }
81
82    /// Attaches the extension.
83    ///
84    /// Attaches the extension to the program and function name specified at load time,
85    /// effectively replacing the original target function.
86    ///
87    /// The returned value can be used to detach the extension and restore the
88    /// original function, see [Extension::detach].
89    pub fn attach(&mut self) -> Result<ExtensionLinkId, ProgramError> {
90        let prog_fd = self.fd()?;
91        let prog_fd = prog_fd.as_fd();
92        let target_fd = self
93            .data
94            .attach_prog_fd
95            .as_ref()
96            .ok_or(ProgramError::NotLoaded)?;
97        let target_fd = target_fd.as_fd();
98        let btf_id = self.data.attach_btf_id.ok_or(ProgramError::NotLoaded)?;
99        // the attach type must be set as 0, which is bpf_attach_type::BPF_CGROUP_INET_INGRESS
100        let link_fd = bpf_link_create(
101            prog_fd,
102            LinkTarget::Fd(target_fd),
103            BPF_CGROUP_INET_INGRESS,
104            Some(btf_id),
105            0,
106            None,
107        )
108        .map_err(|(_, io_error)| SyscallError {
109            call: "bpf_link_create",
110            io_error,
111        })?;
112        self.data
113            .links
114            .insert(ExtensionLink::new(FdLink::new(link_fd)))
115    }
116
117    /// Attaches the extension to another program.
118    ///
119    /// Attaches the extension to a program and/or function other than the one provided
120    /// at load time. You may only attach to another program/function if the BTF
121    /// type signature is identical to that which was verified on load. Attempting to
122    /// attach to an invalid program/function will result in an error.
123    ///
124    /// Once attached, the extension effectively replaces the original target function.
125    ///
126    /// The returned value can be used to detach the extension and restore the
127    /// original function, see [Extension::detach].
128    pub fn attach_to_program(
129        &mut self,
130        program: &ProgramFd,
131        func_name: &str,
132    ) -> Result<ExtensionLinkId, ProgramError> {
133        let target_fd = program.as_fd();
134        let (_, btf_id) = get_btf_info(target_fd, func_name)?;
135        let prog_fd = self.fd()?;
136        let prog_fd = prog_fd.as_fd();
137        // the attach type must be set as 0, which is bpf_attach_type::BPF_CGROUP_INET_INGRESS
138        let link_fd = bpf_link_create(
139            prog_fd,
140            LinkTarget::Fd(target_fd),
141            BPF_CGROUP_INET_INGRESS,
142            Some(btf_id),
143            0,
144            None,
145        )
146        .map_err(|(_, io_error)| SyscallError {
147            call: "bpf_link_create",
148            io_error,
149        })?;
150        self.data
151            .links
152            .insert(ExtensionLink::new(FdLink::new(link_fd)))
153    }
154
155    /// Detaches the extension.
156    ///
157    /// Detaching restores the original code overridden by the extension program.
158    /// See [Extension::attach].
159    pub fn detach(&mut self, link_id: ExtensionLinkId) -> Result<(), ProgramError> {
160        self.data.links.remove(link_id)
161    }
162
163    /// Takes ownership of the link referenced by the provided link_id.
164    ///
165    /// The link will be detached on `Drop` and the caller is now responsible
166    /// for managing its lifetime.
167    pub fn take_link(&mut self, link_id: ExtensionLinkId) -> Result<ExtensionLink, ProgramError> {
168        self.data.take_link(link_id)
169    }
170}

172/// Retrieves the FD of the BTF object for the provided `prog_fd` and the BTF ID of the function
173/// with the name `func_name` within that BTF object.
174fn get_btf_info(
175    prog_fd: BorrowedFd<'_>,
176    func_name: &str,
177) -> Result<(crate::MockableFd, u32), ProgramError> {
178    // retrieve program information
179    let info = sys::bpf_prog_get_info_by_fd(prog_fd, &mut [])?;
180
181    // btf_id refers to the ID of the program btf that was loaded with bpf(BPF_BTF_LOAD)
182    if info.btf_id == 0 {
183        return Err(ProgramError::ExtensionError(ExtensionError::NoBTF));
184    }
185
186    // the bpf fd of the BTF object
187    let btf_fd = sys::bpf_btf_get_fd_by_id(info.btf_id)?;
188
189    // we need to read the btf bytes into a buffer but we don't know the size ahead of time.
190    // assume 4kb. if this is too small we can resize based on the size obtained in the response.
191    let mut buf = vec![0u8; 4096];
192    loop {
193        let info = sys::btf_obj_get_info_by_fd(btf_fd.as_fd(), &mut buf)?;
194        let btf_size = info.btf_size as usize;
195        if btf_size > buf.len() {
196            buf.resize(btf_size, 0u8);
197            continue;
198        }
199        buf.truncate(btf_size);
200        break;
201    }
202
203    let btf = Btf::parse(&buf, Endianness::default()).map_err(ProgramError::Btf)?;
204
205    let btf_id = btf
206        .id_by_type_name_kind(func_name, BtfKind::Func)
207        .map_err(ProgramError::Btf)?;
208
209    Ok((btf_fd, btf_id))
210}
