14/// A Bloom Filter.
15///
16/// # Minimum kernel version
17///
18/// The minimum kernel version required to use this feature is 5.16.
19///
20/// # Examples
21///
22/// ```no_run
23/// # let mut bpf = aya::Ebpf::load(&[])?;
24/// use aya::maps::bloom_filter::BloomFilter;
25///
26/// let mut bloom_filter = BloomFilter::try_from(bpf.map_mut("BLOOM_FILTER").unwrap())?;
27///
28/// bloom_filter.insert(1, 0)?;
29///
30/// assert!(bloom_filter.contains(&1, 0).is_ok());
31/// assert!(bloom_filter.contains(&2, 0).is_err());
32///
33/// # Ok::<(), aya::EbpfError>(())
34/// ```
35#[doc(alias = "BPF_MAP_TYPE_BLOOM_FILTER")]
36#[derive(Debug)]
37pub struct BloomFilter<T, V: Pod> {
38    pub(crate) inner: T,
39    _v: PhantomData<V>,
40}
41
42impl<T: Borrow<MapData>, V: Pod> BloomFilter<T, V> {
43    pub(crate) fn new(map: T) -> Result<Self, MapError> {
44        let data = map.borrow();
45        check_v_size::<V>(data)?;
46
47        Ok(Self {
48            inner: map,
49            _v: PhantomData,
50        })
51    }
52
53    /// Query the existence of the element.
54    pub fn contains(&self, mut value: &V, flags: u64) -> Result<(), MapError> {
55        let fd = self.inner.borrow().fd().as_fd();
56
57        bpf_map_lookup_elem_ptr::<u32, _>(fd, None, &mut value, flags)
58            .map_err(|(_, io_error)| SyscallError {
59                call: "bpf_map_lookup_elem",
60                io_error,
61            })?
62            .ok_or(MapError::ElementNotFound)?;
63        Ok(())
64    }
65}
66
67impl<T: BorrowMut<MapData>, V: Pod> BloomFilter<T, V> {
68    /// Inserts a value into the map.
69    pub fn insert(&mut self, value: impl Borrow<V>, flags: u64) -> Result<(), MapError> {
70        let fd = self.inner.borrow_mut().fd().as_fd();
71        bpf_map_push_elem(fd, value.borrow(), flags).map_err(|(_, io_error)| SyscallError {
72            call: "bpf_map_push_elem",
73            io_error,
74        })?;
75        Ok(())
76    }
77}
