16/// Similar to [`HashMap`](crate::maps::HashMap) but each CPU holds a separate value for a given key. Typically used to
17/// minimize lock contention in eBPF programs.
18///
19/// This type can be used with eBPF maps of type `BPF_MAP_TYPE_PERCPU_HASH` and
20/// `BPF_MAP_TYPE_LRU_PERCPU_HASH`.
21///
22/// # Minimum kernel version
23///
24/// The minimum kernel version required to use this feature is 4.6.
25///
26/// # Examples
27///
28/// ```no_run
29/// # let mut bpf = aya::Ebpf::load(&[])?;
30/// use aya::maps::PerCpuHashMap;
31///
32/// const CPU_IDS: u8 = 1;
33/// const WAKEUPS: u8 = 2;
34///
35/// let mut hm = PerCpuHashMap::<_, u8, u32>::try_from(bpf.map_mut("PER_CPU_STORAGE").unwrap())?;
36/// let cpu_ids = unsafe { hm.get(&CPU_IDS, 0)? };
37/// let wakeups = unsafe { hm.get(&WAKEUPS, 0)? };
38/// for (cpu_id, wakeups) in cpu_ids.iter().zip(wakeups.iter()) {
39///     println!("cpu {} woke up {} times", cpu_id, wakeups);
40/// }
41/// # Ok::<(), aya::EbpfError>(())
42/// ```
43#[doc(alias = "BPF_MAP_TYPE_LRU_PERCPU_HASH")]
44#[doc(alias = "BPF_MAP_TYPE_PERCPU_HASH")]
45pub struct PerCpuHashMap<T, K: Pod, V: Pod> {
46    pub(crate) inner: T,
47    _k: PhantomData<K>,
48    _v: PhantomData<V>,
49}
50
51impl<T: Borrow<MapData>, K: Pod, V: Pod> PerCpuHashMap<T, K, V> {
52    pub(crate) fn new(map: T) -> Result<Self, MapError> {
53        let data = map.borrow();
54        check_kv_size::<K, V>(data)?;
55
56        Ok(Self {
57            inner: map,
58            _k: PhantomData,
59            _v: PhantomData,
60        })
61    }
62
63    /// Returns a slice of values - one for each CPU - associated with the key.
64    pub fn get(&self, key: &K, flags: u64) -> Result<PerCpuValues<V>, MapError> {
65        let fd = self.inner.borrow().fd().as_fd();
66        let values =
67            bpf_map_lookup_elem_per_cpu(fd, key, flags).map_err(|(_, io_error)| SyscallError {
68                call: "bpf_map_lookup_elem",
69                io_error,
70            })?;
71        values.ok_or(MapError::KeyNotFound)
72    }
73
74    /// An iterator visiting all key-value pairs in arbitrary order. The
75    /// iterator item type is `Result<(K, PerCpuValues<V>), MapError>`.
76    pub fn iter(&self) -> MapIter<'_, K, PerCpuValues<V>, Self> {
77        MapIter::new(self)
78    }
79
80    /// An iterator visiting all keys in arbitrary order. The iterator element
81    /// type is `Result<K, MapError>`.
82    pub fn keys(&self) -> MapKeys<'_, K> {
83        MapKeys::new(self.inner.borrow())
84    }
85}
86
87impl<T: BorrowMut<MapData>, K: Pod, V: Pod> PerCpuHashMap<T, K, V> {
88    /// Inserts a slice of values - one for each CPU - for the given key.
89    ///
90    /// # Examples
91    ///
92    /// ```no_run
93    /// # #[derive(thiserror::Error, Debug)]
94    /// # enum Error {
95    /// #     #[error(transparent)]
96    /// #     IO(#[from] std::io::Error),
97    /// #     #[error(transparent)]
98    /// #     Map(#[from] aya::maps::MapError),
99    /// #     #[error(transparent)]
100    /// #     Ebpf(#[from] aya::EbpfError)
101    /// # }
102    /// # let mut bpf = aya::Ebpf::load(&[])?;
103    /// use aya::maps::{PerCpuHashMap, PerCpuValues};
104    /// use aya::util::nr_cpus;
105    ///
106    /// const RETRIES: u8 = 1;
107    ///
108    /// let nr_cpus = nr_cpus().map_err(|(_, error)| error)?;
109    /// let mut hm = PerCpuHashMap::<_, u8, u32>::try_from(bpf.map_mut("PER_CPU_STORAGE").unwrap())?;
110    /// hm.insert(
111    ///     RETRIES,
112    ///     PerCpuValues::try_from(vec![3u32; nr_cpus])?,
113    ///     0,
114    /// )?;
115    /// # Ok::<(), Error>(())
116    /// ```
117    pub fn insert(
118        &mut self,
119        key: impl Borrow<K>,
120        values: PerCpuValues<V>,
121        flags: u64,
122    ) -> Result<(), MapError> {
123        let fd = self.inner.borrow_mut().fd().as_fd();
124        bpf_map_update_elem_per_cpu(fd, key.borrow(), &values, flags).map_err(
125            |(_, io_error)| SyscallError {
126                call: "bpf_map_update_elem",
127                io_error,
128            },
129        )?;
130
131        Ok(())
132    }
133
134    /// Removes a key from the map.
135    pub fn remove(&mut self, key: &K) -> Result<(), MapError> {
136        hash_map::remove(self.inner.borrow_mut(), key)
137    }
138}

140impl<T: Borrow<MapData>, K: Pod, V: Pod> IterableMap<K, PerCpuValues<V>>
141    for PerCpuHashMap<T, K, V>
142{
143    fn map(&self) -> &MapData {
144        self.inner.borrow()
145    }
146
147    fn get(&self, key: &K) -> Result<PerCpuValues<V>, MapError> {
148        Self::get(self, key, 0)
149    }
150}

168    fn test_try_from_ok_lru() {
169        let map_data =
170            || test_utils::new_map(test_utils::new_obj_map::<u32>(BPF_MAP_TYPE_LRU_PERCPU_HASH));
171        let map = Map::PerCpuHashMap(map_data());
172        assert!(PerCpuHashMap::<_, u32, u32>::try_from(&map).is_ok());
173        let map = Map::PerCpuLruHashMap(map_data());
174        assert!(PerCpuHashMap::<_, u32, u32>::try_from(&map).is_ok())
