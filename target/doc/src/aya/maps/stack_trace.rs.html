17/// A hash map of kernel or user space stack traces.
18///
19/// Stack trace maps can be used to store stack traces captured by eBPF programs, which can be
20/// useful for profiling, to associate a trace to an event, etc. You can capture traces calling
21/// `stack_id = bpf_get_stackid(ctx, map, flags)` from eBPF, and then you can retrieve the traces
22/// from their stack ids.
23///
24/// # Minimum kernel version
25///
26/// The minimum kernel version required to use this feature is 4.6.
27///  
28/// # Examples
29///
30/// ```no_run
31/// # #[derive(thiserror::Error, Debug)]
32/// # enum Error {
33/// #     #[error(transparent)]
34/// #     IO(#[from] std::io::Error),
35/// #     #[error(transparent)]
36/// #     Map(#[from] aya::maps::MapError),
37/// #     #[error(transparent)]
38/// #     Ebpf(#[from] aya::EbpfError)
39/// # }
40/// # let bpf = aya::Ebpf::load(&[])?;
41/// use aya::maps::StackTraceMap;
42/// use aya::util::kernel_symbols;
43///
44/// let mut stack_traces = StackTraceMap::try_from(bpf.map("STACK_TRACES").unwrap())?;
45/// // load kernel symbols from /proc/kallsyms
46/// let ksyms = kernel_symbols()?;
47///
48/// // NOTE: you typically send stack_ids from eBPF to user space using other maps
49/// let stack_id = 1234;
50/// let mut stack_trace = stack_traces.get(&stack_id, 0)?;
51///
52/// // here we resolve symbol names using kernel symbols. If this was a user space stack (for
53/// // example captured from a uprobe), you'd have to load the symbols using some other mechanism
54/// // (eg loading the target binary debuginfo)
55/// for frame in stack_trace.frames() {
56///     if let Some(sym) = ksyms.range(..=frame.ip).next_back().map(|(_, s)| s) {
57///         println!(
58///             "{:#x} {}",
59///             frame.ip,
60///             sym
61///         );
62///     } else {
63///         println!(
64///             "{:#x}",
65///             frame.ip
66///         );
67///     }
68/// }
69///
70/// # Ok::<(), Error>(())
71/// ```
72///
73#[derive(Debug)]
74#[doc(alias = "BPF_MAP_TYPE_STACK_TRACE")]
75pub struct StackTraceMap<T> {
76    pub(crate) inner: T,
77    max_stack_depth: usize,
78}
79
80impl<T: Borrow<MapData>> StackTraceMap<T> {
81    pub(crate) fn new(map: T) -> Result<Self, MapError> {
82        let data = map.borrow();
83        let expected = mem::size_of::<u32>();
84        let size = data.obj.key_size() as usize;
85        if size != expected {
86            return Err(MapError::InvalidKeySize { size, expected });
87        }
88
89        let max_stack_depth =
90            sysctl::<usize>("kernel/perf_event_max_stack").map_err(|io_error| SyscallError {
91                call: "sysctl",
92                io_error,
93            })?;
94        let size = data.obj.value_size() as usize;
95        if size > max_stack_depth * mem::size_of::<u64>() {
96            return Err(MapError::InvalidValueSize { size, expected });
97        }
98
99        Ok(Self {
100            inner: map,
101            max_stack_depth,
102        })
103    }
104
105    /// Returns the stack trace with the given stack_id.
106    ///
107    /// # Errors
108    ///
109    /// Returns [`MapError::KeyNotFound`] if there is no stack trace with the
110    /// given `stack_id`, or [`MapError::SyscallError`] if `bpf_map_lookup_elem` fails.
111    pub fn get(&self, stack_id: &u32, flags: u64) -> Result<StackTrace, MapError> {
112        let fd = self.inner.borrow().fd().as_fd();
113
114        let mut frames = vec![0; self.max_stack_depth];
115        bpf_map_lookup_elem_ptr(fd, Some(stack_id), frames.as_mut_ptr(), flags)
116            .map_err(|(_, io_error)| SyscallError {
117                call: "bpf_map_lookup_elem",
118                io_error,
119            })?
120            .ok_or(MapError::KeyNotFound)?;
121
122        let frames = frames
123            .into_iter()
124            .take_while(|ip| *ip != 0)
125            .map(|ip| StackFrame { ip })
126            .collect::<Vec<_>>();
127
128        Ok(StackTrace {
129            id: *stack_id,
130            frames,
131        })
132    }
133
134    /// An iterator visiting all (`stack_id`, `stack_trace`) pairs in arbitrary order. The
135    /// iterator item type is `Result<(u32, StackTrace), MapError>`.
136    pub fn iter(&self) -> MapIter<'_, u32, StackTrace, Self> {
137        MapIter::new(self)
138    }
139
140    /// An iterator visiting all the stack_ids in arbitrary order. The iterator element
141    /// type is `Result<u32, MapError>`.
142    pub fn stack_ids(&self) -> MapKeys<'_, u32> {
143        MapKeys::new(self.inner.borrow())
144    }
145}
146
147impl<T: Borrow<MapData>> IterableMap<u32, StackTrace> for StackTraceMap<T> {
148    fn map(&self) -> &MapData {
149        self.inner.borrow()
150    }
151
152    fn get(&self, index: &u32) -> Result<StackTrace, MapError> {
153        self.get(index, 0)
154    }
155}
156
157impl<'a, T: Borrow<MapData>> IntoIterator for &'a StackTraceMap<T> {
158    type Item = Result<(u32, StackTrace), MapError>;
159    type IntoIter = MapIter<'a, u32, StackTrace, StackTraceMap<T>>;
160
161    fn into_iter(self) -> Self::IntoIter {
162        self.iter()
163    }
164}
165
166impl<T: BorrowMut<MapData>> StackTraceMap<T> {
167    /// Removes the stack trace with the given stack_id.
168    pub fn remove(&mut self, stack_id: &u32) -> Result<(), MapError> {
169        let fd = self.inner.borrow().fd().as_fd();
170        bpf_map_delete_elem(fd, stack_id)
171            .map(|_| ())
172            .map_err(|(_, io_error)| {
173                SyscallError {
174                    call: "bpf_map_delete_elem",
175                    io_error,
176                }
177                .into()
178            })
179    }
180}

182/// A kernel or user space stack trace.
183///
184/// See the [`StackTraceMap`] documentation for examples.
185pub struct StackTrace {
186    /// The stack trace id as returned by `bpf_get_stackid()`.
187    pub id: u32,
188    frames: Vec<StackFrame>,
189}
190
191impl StackTrace {
192    /// Returns the frames in this stack trace.
193    pub fn frames(&self) -> &[StackFrame] {
194        &self.frames
195    }
196}
197
198/// A stack frame.
199pub struct StackFrame {
200    /// The instruction pointer of this frame.
201    pub ip: u64,
202}

204fn sysctl<T: FromStr>(key: &str) -> Result<T, io::Error> {
205    let val = fs::read_to_string(Path::new("/proc/sys").join(key))?;
206    val.trim()
207        .parse::<T>()
208        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, val))
209}
