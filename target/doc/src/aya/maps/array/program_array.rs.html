14/// An array of eBPF program file descriptors used as a jump table.
15///
16/// eBPF programs can jump to other programs calling `bpf_tail_call(ctx,
17/// prog_array, index)`. You can use [`ProgramArray`] to configure which
18/// programs correspond to which jump indexes.
19///
20/// # Minimum kernel version
21///
22/// The minimum kernel version required to use this feature is 4.2.
23///
24/// # Examples
25/// ```no_run
26/// # let mut bpf = aya::Ebpf::load(&[])?;
27/// use aya::maps::ProgramArray;
28/// use aya::programs::CgroupSkb;
29///
30/// let mut prog_array = ProgramArray::try_from(bpf.take_map("JUMP_TABLE").unwrap())?;
31/// let prog_0: &CgroupSkb = bpf.program("example_prog_0").unwrap().try_into()?;
32/// let prog_0_fd =  prog_0.fd().unwrap();
33/// let prog_1: &CgroupSkb = bpf.program("example_prog_1").unwrap().try_into()?;
34/// let prog_1_fd = prog_1.fd().unwrap();
35/// let prog_2: &CgroupSkb = bpf.program("example_prog_2").unwrap().try_into()?;
36/// let prog_2_fd = prog_2.fd().unwrap();
37/// let flags = 0;
38///
39/// // bpf_tail_call(ctx, JUMP_TABLE, 0) will jump to prog_0
40/// prog_array.set(0, &prog_0_fd, flags);
41///
42/// // bpf_tail_call(ctx, JUMP_TABLE, 1) will jump to prog_1
43/// prog_array.set(1, &prog_1_fd, flags);
44///
45/// // bpf_tail_call(ctx, JUMP_TABLE, 2) will jump to prog_2
46/// prog_array.set(2, &prog_2_fd, flags);
47/// # Ok::<(), aya::EbpfError>(())
48/// ```
49#[doc(alias = "BPF_MAP_TYPE_PROG_ARRAY")]
50pub struct ProgramArray<T> {
51    pub(crate) inner: T,
52}
53
54impl<T: Borrow<MapData>> ProgramArray<T> {
55    pub(crate) fn new(map: T) -> Result<Self, MapError> {
56        let data = map.borrow();
57        check_kv_size::<u32, RawFd>(data)?;
58
59        Ok(Self { inner: map })
60    }
61
62    /// An iterator over the indices of the array that point to a program. The iterator item type
63    /// is `Result<u32, MapError>`.
64    pub fn indices(&self) -> MapKeys<'_, u32> {
65        MapKeys::new(self.inner.borrow())
66    }
67}
68
69impl<T: BorrowMut<MapData>> ProgramArray<T> {
70    /// Sets the target program file descriptor for the given index in the jump table.
71    ///
72    /// When an eBPF program calls `bpf_tail_call(ctx, prog_array, index)`, control
73    /// flow will jump to `program`.
74    pub fn set(&mut self, index: u32, program: &ProgramFd, flags: u64) -> Result<(), MapError> {
75        let data = self.inner.borrow_mut();
76        check_bounds(data, index)?;
77        let fd = data.fd().as_fd();
78        let prog_fd = program.as_fd();
79        let prog_fd = prog_fd.as_raw_fd();
80
81        bpf_map_update_elem(fd, Some(&index), &prog_fd, flags).map_err(|(_, io_error)| {
82            SyscallError {
83                call: "bpf_map_update_elem",
84                io_error,
85            }
86        })?;
87        Ok(())
88    }
89
90    /// Clears the value at index in the jump table.
91    ///
92    /// Calling `bpf_tail_call(ctx, prog_array, index)` on an index that has been cleared returns an
93    /// error.
94    pub fn clear_index(&mut self, index: &u32) -> Result<(), MapError> {
95        let data = self.inner.borrow_mut();
96        check_bounds(data, *index)?;
97        let fd = data.fd().as_fd();
98
99        bpf_map_delete_elem(fd, index)
100            .map(|_| ())
101            .map_err(|(_, io_error)| {
102                SyscallError {
103                    call: "bpf_map_delete_elem",
104                    io_error,
105                }
106                .into()
107            })
108    }
109}
